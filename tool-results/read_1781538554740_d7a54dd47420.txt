   112→        static void initHooks(Window window, HTMLCanvasElement canvaz) {
   113→                win = window;
   114→                canvas = canvaz;
   115→                canvas.getStyle().setProperty("cursor", "default");
   116→                win.addEventListener("contextmenu", contextmenu = new EventListener<MouseEvent>() {
   117→                        @Override
   118→                        public void handleEvent(MouseEvent evt) {
   119→                                evt.preventDefault();
   120→                                evt.stopPropagation();
   121→                        }
   122→                });
   123→                canvas.addEventListener("mousedown", mousedown = new EventListener<MouseEvent>() {
   124→                        @Override
   125→                        public void handleEvent(MouseEvent evt) {
   126→                                evt.preventDefault();
   127→                                evt.stopPropagation();
   128→                                int b = evt.getButton();
   129→                                buttonStates[b == 1 ? 2 : (b == 2 ? 1 : b)] = true;
   130→                                mouseEvents.add(evt);
   131→                        }
   132→                });
   133→                canvas.addEventListener("mouseup", mouseup = new EventListener<MouseEvent>() {
   134→                        @Override
   135→                        public void handleEvent(MouseEvent evt) {
   136→                                evt.preventDefault();
   137→                                evt.stopPropagation();
   138→                                int b = evt.getButton();
   139→                                buttonStates[b == 1 ? 2 : (b == 2 ? 1 : b)] = false;
   140→                                mouseEvents.add(evt);
   141→                        }
   142→                });
   143→                canvas.addEventListener("mousemove", mousemove = new EventListener<MouseEvent>() {
   144→                        @Override
   145→                        public void handleEvent(MouseEvent evt) {
   146→                                evt.preventDefault();
   147→                                evt.stopPropagation();
   148→                                mouseX = (int)(getOffsetX(evt) * win.getDevicePixelRatio());
   149→                                mouseY = (int)((canvas.getClientHeight() - getOffsetY(evt)) * win.getDevicePixelRatio());
   150→                                mouseDX += evt.getMovementX();
   151→                                mouseDY += -evt.getMovementY();
   152→                                if(hasBeenActive()) {
   153→                                        mouseEvents.add(evt);
   154→                                }
   155→                        }
   156→                });
   157→                canvas.addEventListener("mouseenter", mouseenter = new EventListener<MouseEvent>() {
   158→                        @Override
   159→                        public void handleEvent(MouseEvent evt) {
   160→                                isMouseOverWindow = true;
   161→                        }
   162→                });
   163→                canvas.addEventListener("mouseleave", mouseleave = new EventListener<MouseEvent>() {
   164→                        @Override
   165→                        public void handleEvent(MouseEvent evt) {
   166→                                isMouseOverWindow = false;
   167→                        }
   168→                });
   169→                win.addEventListener("keydown", keydown = new EventListener<KeyboardEvent>() {
   170→                        @Override
   171→                        public void handleEvent(KeyboardEvent evt) {
   172→                                int w = getWhich(evt);
   173→                                if (w == 122) { // F11
   174→                                        toggleFullscreen();
   175→                                }
   176→                                evt.preventDefault();
   177→                                evt.stopPropagation();
   178→                                if(!enableRepeatEvents && evt.isRepeat()) return;
   179→                                int ww = processFunctionKeys(w);
   180→                                keyStates[KeyboardConstants.getEaglerKeyFromBrowser(ww, ww == w ? evt.getLocation() : 0)] = true;
   181→                                keyEvents.add(evt);
   182→                        }
   183→                });
   184→                win.addEventListener("keyup", keyup = new EventListener<KeyboardEvent>() {
   185→                        @Override
   186→                        public void handleEvent(KeyboardEvent evt) {
   187→                                int w = getWhich(evt);
   188→                                evt.preventDefault();
   189→                                evt.stopPropagation();
   190→                                if(!enableRepeatEvents && evt.isRepeat()) return;
   191→                                int ww = processFunctionKeys(w);
   192→                                int eagKey = KeyboardConstants.getEaglerKeyFromBrowser(ww, ww == w ? evt.getLocation() : 0);
   193→                                keyStates[eagKey] = false;
   194→                                if(eagKey == functionKeyModifier) {
   195→                                        for(int key = KeyboardConstants.KEY_F1; key <= KeyboardConstants.KEY_F10; ++key) {
   196→                                                keyStates[key] = false;
   197→                                        }
   198→                                }
   199→                                keyEvents.add(evt);
   200→                        }
   201→                });
   202→                win.addEventListener("keypress", keypress = new EventListener<KeyboardEvent>() {
   203→                        @Override
   204→                        public void handleEvent(KeyboardEvent evt) {
   205→                                evt.preventDefault();
   206→                                evt.stopPropagation();
   207→                                if(enableRepeatEvents && evt.isRepeat()) keyEvents.add(evt);
   208→                        }
   209→                });
   210→                canvas.addEventListener("wheel", wheel = new EventListener<WheelEvent>() {
   211→                        @Override
   212→                        public void handleEvent(WheelEvent evt) {
   213→                                evt.preventDefault();
   214→                                evt.stopPropagation();
   215→                                mouseEvents.add(evt);
   216→                                mouseDWheel += evt.getDeltaY();
   217→                        }
   218→                });
   219→                win.addEventListener("blur", new EventListener<WheelEvent>() {
   220→                        @Override
   221→                        public void handleEvent(WheelEvent evt) {
   222→                                isWindowFocused = false;
   223→                                for(int i = 0; i < buttonStates.length; ++i) {
   224→                                        buttonStates[i] = false;
   225→                                }
   226→                                for(int i = 0; i < keyStates.length; ++i) {
   227→                                        keyStates[i] = false;
   228→                                }
   229→                        }
   230→                });
   231→                win.addEventListener("focus", new EventListener<WheelEvent>() {
   232→                        @Override
   233→                        public void handleEvent(WheelEvent evt) {
   234→                                isWindowFocused = true;
   235→                        }
   236→                });
   237→                win.getDocument().addEventListener("pointerlockchange", pointerlock = new EventListener<WheelEvent>() {
   238→                        @Override
   239→                        public void handleEvent(WheelEvent evt) {
   240→                                Window.setTimeout(new TimerHandler() {
   241→                                        @Override
   242→                                        public void onTimer() {
   243→                                                boolean grab = isPointerLocked();
   244→                                                if(!grab) {
   245→                                                        if(pointerLockFlag) {
   246→                                                                mouseUngrabTimer = System.currentTimeMillis();
   247→                                                        }
   248→                                                }
   249→                                                pointerLockFlag = grab;
   250→                                        }
   251→                                }, 60);
   252→                                mouseDX = 0.0D;
   253→                                mouseDY = 0.0D;
   254→                        }
   255→                });
   256→
   257→                try {
   258→                        onBeforeCloseRegister();
   259→                }catch(Throwable t) {
   260→                }
   261→
   262→                try {
   263→                        asyncRequestAnimationFrame();
   264→                        vsyncSupport = true;
   265→                }catch(Throwable t) {
   266→                        PlatformRuntime.logger.error("VSync is not supported on this browser!");
   267→                }
   268→
   269→                fullscreenQuery = fullscreenMediaQuery();
   270→                if (keyboardLockSupported = checkKeyboardLockSupported()) {
   271→                        TeaVMUtils.addEventListener(fullscreenQuery, "change", new EventListener<Event>() {
   272→                                @Override
   273→                                public void handleEvent(Event evt) {
   274→                                        if (!mediaQueryMatches(evt)) {
   275→                                                unlockKeys();
   276→                                                lockKeys = false;
   277→                                        }
   278→                                }
   279→                        });
   280→                }
   281→        }

   318→        public static void update() {
   319→                double r = win.getDevicePixelRatio();
   320→                int w = PlatformRuntime.parent.getClientWidth();
   321→                int h = PlatformRuntime.parent.getClientHeight();
   322→                int w2 = windowWidth = (int)(w * r);
   323→                int h2 = windowHeight = (int)(h * r);
   324→                if(canvas.getWidth() != w2) {
   325→                        canvas.setWidth(w2);
   326→                }
   327→                if(canvas.getHeight() != h2) {
   328→                        canvas.setHeight(h2);
   329→                }
   330→                flipBuffer();
   331→                if (PlatformRuntime.recording) {
   332→                        long t = System.currentTimeMillis();
   333→                        if(t - PlatformRuntime.lastFrame > (1000 / 30)) {
   334→                                PlatformRuntime.recFrame();
   335→                                PlatformRuntime.lastFrame = t;
   336→                        }
   337→                }
   338→                if(getVisibilityState(win.getDocument())) {
   339→                        if(vsyncSupport && vsync) {
   340→                                asyncRequestAnimationFrame();
   341→                        }else {
   342→                                EagUtils.sleep(0l);
   343→                        }
   344→                }else {
   345→                        EagUtils.sleep(50l);
   346→                }
   347→        }

   352→        private static void asyncRequestAnimationFrame(AsyncCallback<Void> cb) {
   353→                final boolean[] hasCompleted = new boolean[1];
   354→                final int[] timeout = new int[] { -1 };
   355→                Window.requestAnimationFrame((d) -> {
   356→                        if(!hasCompleted[0]) {
   357→                                hasCompleted[0] = true;
   358→                                Window.clearTimeout(timeout[0]);
   359→                                cb.complete(null);
   360→                        }
   361→                });
   362→                timeout[0] = Window.setTimeout(() -> {
   363→                        if(!hasCompleted[0]) {
   364→                                hasCompleted[0] = true;
   365→                                cb.complete(null);
   366→                        }
   367→                }, 50);
   368→        }

   374→        static void initFramebuffer(WebGL2RenderingContext ctx, WebGLFramebuffer fbo, int sw, int sh) {
   375→                context = ctx;
   376→                mainFramebuffer = fbo;
   377→                
   378→                framebufferWidth = windowWidth = sw;
   379→                framebufferHeight = windowHeight = sh;
   380→                
   381→                ctx.bindFramebuffer(FRAMEBUFFER, fbo);
   382→
   383→                mainColorRenderbuffer = ctx.createRenderbuffer();
   384→                mainDepthRenderbuffer = ctx.createRenderbuffer();
   385→                
   386→                ctx.bindRenderbuffer(RENDERBUFFER, mainColorRenderbuffer);
   387→                ctx.renderbufferStorage(RENDERBUFFER, RGBA8, sw, sh);
   388→                ctx.framebufferRenderbuffer(FRAMEBUFFER, COLOR_ATTACHMENT0, RENDERBUFFER, mainColorRenderbuffer);
   389→                
   390→                ctx.bindRenderbuffer(RENDERBUFFER, mainDepthRenderbuffer);
   391→                ctx.renderbufferStorage(RENDERBUFFER, DEPTH_COMPONENT32F, sw, sh);
   392→                ctx.framebufferRenderbuffer(FRAMEBUFFER, DEPTH_ATTACHMENT, RENDERBUFFER, mainDepthRenderbuffer);
   393→                
   394→                ctx.drawBuffers(new int[] { COLOR_ATTACHMENT0 });
   395→        }

   397→        private static void flipBuffer() {
   398→                
   399→                context.bindFramebuffer(READ_FRAMEBUFFER, mainFramebuffer);
   400→                context.bindFramebuffer(DRAW_FRAMEBUFFER, null);
   401→                context.blitFramebuffer(0, 0, framebufferWidth, framebufferHeight, 0, 0, windowWidth, windowHeight, COLOR_BUFFER_BIT, NEAREST);
   402→                
   403→                context.bindFramebuffer(FRAMEBUFFER, mainFramebuffer);
   404→                
   405→                if(windowWidth != framebufferWidth || windowHeight != framebufferHeight) {
   406→                        framebufferWidth = windowWidth;
   407→                        framebufferHeight = windowHeight;
   408→                        
   409→                        context.bindRenderbuffer(RENDERBUFFER, mainColorRenderbuffer);
   410→                        context.renderbufferStorage(RENDERBUFFER, RGBA8, framebufferWidth, framebufferHeight);
   411→                        
   412→                        context.bindRenderbuffer(RENDERBUFFER, mainDepthRenderbuffer);
   413→                        context.renderbufferStorage(RENDERBUFFER, DEPTH_COMPONENT32F, framebufferWidth, framebufferHeight);
   414→                }

   523→        public static void mouseSetGrabbed(boolean grab) {
   524→                long t = System.currentTimeMillis();
   525→                pointerLockFlag = grab;
   526→                mouseGrabTimer = t;
   527→                if(grab) {
   528→                        canvas.requestPointerLock();
   529→                        if(mouseUngrabTimeout != -1) Window.clearTimeout(mouseUngrabTimeout);
   530→                        mouseUngrabTimeout = -1;
   531→                        if(t - mouseUngrabTimer < 3000l) {
   532→                                mouseUngrabTimeout = Window.setTimeout(new TimerHandler() {
   533→                                        @Override
   534→                                        public void onTimer() {
   535→                                                canvas.requestPointerLock();
   536→                                        }
   537→                                }, 3100 - (int)(t - mouseUngrabTimer));
   538→                        }
   539→                }else {
   540→                        if(mouseUngrabTimeout != -1) Window.clearTimeout(mouseUngrabTimeout);
   541→                        mouseUngrabTimeout = -1;
   542→                        Window.current().getDocument().exitPointerLock();
   543→                }
   544→                mouseDX = 0.0D;
   545→                mouseDY = 0.0D;
   546→        }
