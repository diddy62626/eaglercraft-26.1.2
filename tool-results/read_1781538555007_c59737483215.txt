    86→        public static void create() {
    87→                win = Window.current();
    88→                doc = win.getDocument();
    89→                DebugConsoleWindow.initialize(win);
    90→                
    91→                logger.info("Creating main game canvas");
    92→                
    93→                parent = doc.getElementById(ClientMain.configRootElementId);
    94→                if(parent == null) {
    95→                        throw new RuntimeInitializationFailureException("Root element \"" + ClientMain.configRootElementId + "\" was not found in this document!");
    96→                }
    97→
    98→                CSSStyleDeclaration style = parent.getStyle();
    99→                style.setProperty("overflowX", "hidden");
   100→                style.setProperty("overflowY", "hidden");
   101→                
   102→                canvas = (HTMLCanvasElement) doc.createElement("canvas");
   103→                
   104→                style = canvas.getStyle();
   105→                style.setProperty("width", "100%");
   106→                style.setProperty("height", "100%");
   107→                style.setProperty("image-rendering", "pixelated");
   108→                
   109→                double r = win.getDevicePixelRatio();
   110→                int iw = parent.getClientWidth();
   111→                int ih = parent.getClientHeight();
   112→                int sw = (int)(r * iw);
   113→                int sh = (int)(r * ih);
   114→
   115→                canvas.setWidth(sw);
   116→                canvas.setHeight(sh);
   117→                
   118→                parent.appendChild(canvas);
   119→                
   120→                try {
   121→                        PlatformInput.initHooks(win, canvas);
   122→                }catch(Throwable t) {
   123→                        throw new RuntimeInitializationFailureException("Exception while registering window event handlers", t);
   124→                }
   125→                
   126→                try {
   127→                        doc.exitPointerLock();
   128→                }catch(Throwable t) {
   129→                        throw new PlatformIncompatibleException("Mouse cursor lock is not available on this device!");
   130→                }
   131→
   132→                logger.info("Creating WebGL context");
   133→                
   134→                JSObject webgl_ = canvas.getContext("webgl2", youEagler());
   135→                if(webgl_ == null) {
   136→                        throw new PlatformIncompatibleException("WebGL 2.0 is not supported on this device!");
   137→                }
   138→                
   139→                webgl = (WebGL2RenderingContext) webgl_;
   140→                PlatformOpenGL.setCurrentContext(webgl);
   141→                
   142→                mainFramebuffer = webgl.createFramebuffer();
   143→                PlatformInput.initFramebuffer(webgl, mainFramebuffer, sw, sh);
   144→                
   145→                EarlyLoadScreen.paintScreen();
   146→                
   147→                EPKFileEntry[] epkFiles = ClientMain.configEPKFiles;
   148→                
   149→                for(int i = 0; i < epkFiles.length; ++i) {
   150→                        String url = epkFiles[i].url;
   151→                        String logURL = url.startsWith("data:") ? "<data: " + url.length() + " chars>" : url;
   152→                        
   153→                        logger.info("Downloading: {}", logURL);
   154→                        
   155→                        ArrayBuffer epkFileData = downloadRemoteURI(url);
   156→                        
   157→                        if(epkFileData == null) {
   158→                                throw new RuntimeInitializationFailureException("Could not download EPK file \"" + url + "\"");
   159→                        }
   160→                        
   161→                        logger.info("Decompressing: {}", logURL);
   162→                        
   163→                        try {
   164→                                EPKLoader.loadEPK(epkFileData, epkFiles[i].path, PlatformAssets.assets);
   165→                        }catch(Throwable t) {
   166→                                throw new RuntimeInitializationFailureException("Could not extract EPK file \"" + url + "\"", t);
   167→                        }
   168→                }
   169→
   170→                logger.info("Loaded {} resources from EPKs", PlatformAssets.assets.size());
   171→
   172→                byte[] finalLoadScreen = PlatformAssets.getResourceBytes("/assets/eagler/eagtek.png");
   173→
   174→                logger.info("Initializing filesystem...");
   175→
   176→                try {
   177→                        PlatformFilesystem.initialize(getClientConfigAdapter().getResourcePacksDB());
   178→                        EaglerFolderResourcePack.setSupported(true);
   179→                }catch(FilesystemDatabaseLockedException t) {
   180→                        logger.error("Could not initialize filesystem, database is locked!");
   181→                }catch(Throwable t) {
   182→                        logger.error("Could not initialize filesystem, encountered an exception!");
   183→                        logger.error(t);
   184→                }
   185→
   186→                if(!EaglerFolderResourcePack.isSupported()) {
   187→                        logger.error("Resource packs will be disabled for this session");
   188→                }
   189→
   190→                logger.info("Initializing sound engine...");
   191→
   192→                PlatformInput.pressAnyKeyScreen();
   193→
   194→                PlatformAudio.initialize();
   195→
   196→                if(finalLoadScreen != null) {
   197→                        EarlyLoadScreen.paintFinal(finalLoadScreen);
   198→                }
   199→
   200→                EarlyLoadScreen.destroy();
   201→
   202→                logger.info("Platform initialization complete");
   203→
   204→                FixWebMDurationJS.checkOldScriptStillLoaded();
   205→        }

   338→        private static void downloadRemoteURIOld(String assetPackageURI, final AsyncCallback<ArrayBuffer> cb) {
   339→                final XMLHttpRequest request = XMLHttpRequest.create();
   340→                request.setResponseType("arraybuffer");
   341→                request.open("GET", assetPackageURI, true);
   342→                
   343→                TeaVMUtils.addEventListener(request, "load", new EventListener<Event>() {
   344→                        @Override
   345→                        public void handleEvent(Event evt) {
   346→                                int stat = request.getStatus();
   347→                                if(stat == 0 || (stat >= 200 && stat < 400)) {
   348→                                        cb.complete((ArrayBuffer)request.getResponse());
   349→                                }else {
   350→                                        cb.complete(null);
   351→                                }
   352→                        }
   353→                });
   354→                
   355→                TeaVMUtils.addEventListener(request, "error", new EventListener<Event>() {
   356→                        @Override
   357→                        public void handleEvent(Event evt) {
   358→                                cb.complete(null);
   359→                        }
   360→                });
   361→                
   362→                request.send();
   363→        }

   380→        public static native ArrayBuffer downloadRemoteURI(final String assetPackageURI, final boolean forceCache);
   381→
   382→        private static void downloadRemoteURI(final String assetPackageURI, final boolean useCache, final AsyncCallback<ArrayBuffer> cb) {
   383→                doFetchDownload(assetPackageURI, useCache ? "force-cache" : "no-store",
   384→                                assetPackageURI.startsWith("data:application/octet-stream;base64,") ? (data) -> {
   385→                                        if(data != null) {
   386→                                                cb.complete(data);
   387→                                        }else {
   388→                                                logger.error("Caught an error decoding base64 via fetch, doing it the slow way instead...");
   389→                                                byte[] b = null;
   390→                                                try {
   391→                                                        b = Base64.decodeBase64(assetPackageURI.substring(37));
   392→                                                }catch(Throwable t) {
   393→                                                        logger.error("Failed to manually decode base64!", t);
   394→                                                        cb.complete(null);
   395→                                                        return;
   396→                                                }
   397→                                                cb.complete(TeaVMUtils.unwrapArrayBuffer(b));
   398→                                        }
   399→                                } : cb::complete);
   400→        }

   417→        public static void getStackTrace(Throwable t, Consumer<String> ret) {
   418→                JSObject o = JSExceptions.getJSException(t);
   419→                if(o != null) {
   420→                        try {
   421→                                JSError err = o.cast();
   422→                                String stack = err.getStack();
   423→                                if(stack != null) {
   424→                                        String[] stackElements = stack.split("[\\r\\n]+");
   425→                                        if(stackElements.length > 0) {
   426→                                                for(int i = 0; i < stackElements.length; ++i) {
   427→                                                        String str = stackElements[i].trim();
   428→                                                        if(str.startsWith("at ")) {
   429→                                                                str = str.substring(3).trim();
   430→                                                        }
   431→                                                        ret.accept(str);
   432→                                                }
   433→                                                return;
   434→                                        }
   435→                                }
   436→                        }catch(Throwable tt) {
   437→                                ret.accept("[ error: " + t.toString() + " ]");
   438→                        }
   439→                }
   440→                getFallbackStackTrace(t, ret);
   441→        }
   442→        
   443→        private static void getFallbackStackTrace(Throwable t, Consumer<String> ret) {
   444→                StackTraceElement[] el = t.getStackTrace();
   445→                if(el.length > 0) {
   446→                        for(int i = 0; i < el.length; ++i) {
   447→                                ret.accept(el[i].toString());
   448→                        }
   449→                }else {
   450→                        ret.accept("[no stack trace]");
   451→                }
   452→        }
