   586→        private static final void checkErr(String name) {
   587→                int i = ctx.getError();
   588→                if(i != 0) {
   589→                        logger.error("########## GL ERROR ##########");
   590→                        logger.error("@ {}", name);
   591→                        do {
   592→                                logger.error("#{} - {}", i, EaglercraftGPU.gluErrorString(i));
   593→                        }while((i = ctx.getError()) != 0);
   594→                        try {
   595→                                throw new RuntimeException("GL Error Detected!");
   596→                        }catch(Throwable t) {
   597→                                logger.log(Level.ERROR, t);
   598→                        }
   599→                        logger.error("##############################");
   600→                }
   601→        }
