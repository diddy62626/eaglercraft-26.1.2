   123→         * masterGain <- musicGain <- compressor <- destination
   124→         * masterGain <- sfxGain   <- compressor <- destination

   130→                        setCompressorThreshold(compressorNode, -24);
   131→                        setCompressorKnee(compressorNode, 30);
   132→                        setCompressorRatio(compressorNode, 12);
   133→                        setCompressorAttack(compressorNode, 0.003);
   134→                        setCompressorRelease(compressorNode, 0.25);
   135→                }
   136→
   137→                // Create gain nodes
   138→                masterGain = createGainNode0();
   139→                musicGain = createGainNode0();
   140→                sfxGain = createGainNode0();
   141→
   142→                // Set initial volumes
   143→                setGainValue(masterGain, masterVolume);
   144→                setGainValue(musicGain, musicVolume);
   145→                setGainValue(sfxGain, sfxVolume);
   146→
   147→                // Connect: musicGain -> masterGain -> compressor -> destination
   148→                connectNodes(musicGain, masterGain);
   149→                connectNodes(sfxGain, masterGain);
   150→
   151→                if (compressorNode != null) {
   152→                        connectNodes(masterGain, compressorNode);
   153→                        connectNodes(compressorNode, getDestination0());
   154→                } else {
   155→                        connectNodes(masterGain, getDestination0());
   156→                }
   157→        }

   232→        /**
   233→         * Creates an AudioBuffer from PCM float data.
   234→         *
   235→         * @param pcm Float PCM data (mono or interleaved stereo)
   236→         * @param channels Number of channels (1 or 2)
   237→         * @param sampleRate Sample rate of the data
   238→         * @return AudioBuffer JSObject, or null on failure
   239→         */
   240→        public static JSObject createAudioBuffer(float[] pcm, int channels, int sampleRate) {
   241→                if (audioContext == null) return null;
   242→
   243→                int frames = pcm.length / channels;
   244→                JSObject buffer = createAudioBuffer0(channels, frames, sampleRate);
   245→                if (buffer == null) return null;
   246→
   247→                // Write channel data
   248→                Float32Array channelData = createFloat32Array(frames);
   249→                for (int ch = 0; ch < channels; ch++) {
   250→                        for (int i = 0; i < frames; i++) {
   251→                                channelData.set(i, pcm[i * channels + ch]);
   252→                        }
   253→                        writeAudioBufferChannel(buffer, ch, channelData);
   254→                }
   255→
   256→                return buffer;
   257→        }

   259→        /**
   260→         * Creates an AudioBuffer from 16-bit PCM data.
   261→         *
   262→         * @param pcm 16-bit PCM data
   263→         * @param channels Number of channels (1 or 2)
   264→         * @param sampleRate Sample rate of the data
   265→         * @return AudioBuffer JSObject, or null on failure
   266→         */
   267→        public static JSObject createAudioBuffer16(Int16Array pcm, int channels, int sampleRate) {
   268→                if (audioContext == null) return null;
   269→
   270→                int frames = pcm.getLength() / channels;
   271→                JSObject buffer = createAudioBuffer0(channels, frames, sampleRate);
   272→                if (buffer == null) return null;
   273→
   274→                // Convert 16-bit to float and write
   275→                for (int ch = 0; ch < channels; ch++) {
   276→                        Float32Array channelData = createFloat32Array(frames);
   277→                        for (int i = 0; i < frames; i++) {
   278→                                short sample = (short) pcm.get(i * channels + ch);
   279→                                channelData.set(i, sample / 32768.0f);
   280→                        }
   281→                        writeAudioBufferChannel(buffer, ch, channelData);
   282→                }
   283→
   284→                return buffer;
   285→        }

   303→
   304→        /**
   305→         * Plays a sound buffer.
   306→         *
   307→         * @param buffer The AudioBuffer to play
   308→         * @param pitch Playback pitch (1.0 = normal)
   309→         * @param volume Volume (0.0 to 1.0)
   310→         * @param type Sound type (SFX, MUSIC, or VOICE)
   311→         * @return Source node JSObject, or null on failure
   312→         */
   313→        public static JSObject playSound(JSObject buffer, float pitch, float volume, int type) {
   314→                if (audioContext == null || buffer == null) return null;
   315→
   316→                JSObject source = createBufferSource0();
   317→                if (source == null) return null;
   318→
   319→                setSourceBuffer(source, buffer);
   320→                setSourcePlaybackRate(source, pitch);
   321→                setSourceLoop(source, false);
   322→
   323→                // Create per-source gain for volume control
   324→                JSObject gain = createGainNode0();
   325→                setGainValue(gain, volume);
   326→
   327→                // Connect: source -> gain -> type-specific gain
   328→                connectNodes(source, gain);
   329→
   330→                switch (type) {
   331→                        case SOUND_TYPE_MUSIC:
   332→                                connectNodes(gain, musicGain);
   333→                                break;
   334→                        case SOUND_TYPE_VOICE:
   335→                                connectNodes(gain, sfxGain); // Voice uses SFX channel
   336→                                break;
   337→                        case SOUND_TYPE_SFX:
   338→                        default:
   339→                                connectNodes(gain, sfxGain);
   340→                                break;
   341→                }
   342→
   343→                // Start playback
   344→                startSource0(source, 0);
   345→
   346→                // Track the source for cleanup
   347→                trackSource(source, gain);
   348→
   349→                return source;
   350→        }

   488→        // ========== Source Tracking ==========
   489→
   490→        private static void trackSource(JSObject source, JSObject gainNode) {
   491→                if (activeSourceCount < MAX_SOURCES) {
   492→                        activeSources[activeSourceCount++] = source;
   493→                }
   494→
   495→                // Set up onended callback for cleanup
   496→                setSourceOnEnded(source, () -> {
   497→                        for (int i = 0; i < activeSourceCount; i++) {
   498→                                if (activeSources[i] == source) {
   499→                                        activeSources[i] = activeSources[--activeSourceCount];
   500→                                        activeSources[activeSourceCount] = null;
   501→                                        break;
   502→                                }
   503→                        }
   504→                });
   505→        }

   508→         * Stops all currently playing sounds.

   510→        public static void stopAllSounds() {
   511→                for (int i = 0; i < activeSourceCount; i++) {
   512→                        if (activeSources[i] != null) {
   513→                                stopSource0(activeSources[i]);
   514→                                activeSources[i] = null;
   515→                        }
   516→                }
   517→                activeSourceCount = 0;
   518→        }
