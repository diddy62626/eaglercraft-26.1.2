     8→/**
     9→ * Simplified integrated server for EaglerCraft 26.1.2 singleplayer.
    10→ *
    11→ * <p>Runs inside a Web Worker and manages the full Minecraft server
    12→ * simulation: world loading/saving to IndexedDB, chunk generation,
    13→ * entity ticking, block ticking, and player management.</p>
    14→ *
    15→ * <h3>World Storage</h3>
    16→ * <p>Worlds are persisted to IndexedDB with the following structure:</p>
    17→ * <ul>
    18→ *   <li>World metadata (seed, gamemode, time, etc.)</li>
    19→ *   <li>Region files (groups of 32x32 chunks in compressed format)</li>
    20→ *   <li>Player data (inventory, health, position, etc.)</li>
    21→ *   <li>Level dat (world settings and preferences)</li>
    22→ * </ul>
    23→ *
    24→ * <h3>Tick Pipeline</h3>
    25→ * <ol>
    26→ *   <li>Process incoming IPC messages from client</li>
    27→ *   <li>Tick player (movement, interaction, inventory)</li>
    28→ *   <li>Tick entities (AI, physics, collision)</li>
    29→ *   <li>Tick block entities (furnaces, hoppers, etc.)</li>
    30→ *   <li>Tick scheduled block updates (redstone, flow, etc.)</li>
    31→ *   <li>Generate/load/unload chunks based on player position</li>
    32→ *   <li>Send render updates to client via IPC</li>
    33→ *   <li>Autosave check (every 600 ticks / 30 seconds)</li>
    34→ * </ol>
    35→ *
    36→ * @author Eaglercraft Team
    37→ * @version 2.6
    38→ * @since 2.6
    39→ */

    93→                final int x;

    95→                final byte[] blockIDs;
    96→                final byte[] blockMetadata;
    97→                final byte[] blockLight;
    98→                final byte[] skyLight;
    99→                boolean modified;
   100→                boolean populated;
   101→
   102→                ChunkData(int x, int z) {
   103→                        this.x = x;
   104→                        this.z = z;
   105→                        int blockCount = CHUNK_SIZE * WORLD_HEIGHT * CHUNK_SIZE;
   106→                        this.blockIDs = new byte[blockCount];
   107→                        this.blockMetadata = new byte[blockCount / 2];
   108→                        this.blockLight = new byte[blockCount / 2];
   109→                        this.skyLight = new byte[blockCount / 2];
   110→                        this.modified = false;
   111→                        this.populated = false;
   112→                }
   113→
   114→                int getBlockIndex(int bx, int by, int bz) {
   115→                        return (by * CHUNK_SIZE + bz) * CHUNK_SIZE + bx;
   116→                }
   117→
   118→                int getBlockID(int bx, int by, int bz) {
   119→                        return blockIDs[getBlockIndex(bx, by, bz)] & 0xFF;

   123→                        blockIDs[getBlockIndex(bx, by, bz)] = (byte) id;

   192→        /**
   193→         * Parses initialization data from the client.
   194→         */
   195→        private void parseInitData(byte[] data) {
   196→                int offset = 0;
   197→                if (data.length < 8) return;
   198→
   199→                // Read world seed (8 bytes)
   200→                worldSeed = 0;
   201→                for (int i = 0; i < 8; i++) {
   202→                        worldSeed = (worldSeed << 8) | (data[offset++] & 0xFFL);
   203→                }
   204→
   205→                // Read game mode (1 byte)
   206→                if (offset < data.length) {
   207→                        playerGameMode = data[offset++] & 0xFF;
   208→                        gameMode = playerGameMode;
   209→                }
   210→
   211→                // Read view distance (1 byte)
   212→                if (offset < data.length) {
   213→                        viewDistance = Math.max(2, Math.min(MAX_CHUNK_RADIUS, data[offset++] & 0xFF));
   214→                }
   215→
   216→                // Read player name (remaining bytes as UTF-8)
   217→                if (offset < data.length) {
   218→                        byte[] nameBytes = new byte[data.length - offset];
   219→                        System.arraycopy(data, offset, nameBytes, 0, nameBytes.length);
   220→                        playerName = new String(nameBytes);
   221→                }
   222→        }

   226→        /**
   227→         * Main server tick. Called 20 times per second.
   228→         */
   229→        public void tick() {
   230→                if (!initialized || paused) return;
   231→                tickCount++;
   232→                worldTime++;
   233→                dayTime = (dayTime + 1) % 24000;
   234→
   235→                // Process pending chunk sends
   236→                processPendingChunkSends();
   237→
   238→                // Tick player
   239→                tickPlayer();
   240→
   241→                // Tick entities
   242→                tickEntities();
   243→
   244→                // Tick scheduled block updates
   245→                tickBlockUpdates();
   246→
   247→                // Check chunk loading/unloading
   248→                updateChunkLoading();
   249→
   250→                // Autosave
   251→                if (tickCount % AUTOSAVE_INTERVAL == 0) {
   252→                        autosave();
   253→                }
   254→
   255→                // Send time update every 20 ticks
   256→                if (tickCount % 20 == 0) {
   257→                        sendTimeUpdate();
   258→                }
   259→        }

   289→        private void updateChunkLoading() {
   290→                int pcx = (int) playerX >> 4;
   291→                int pcz = (int) playerZ >> 4;
   292→
   293→                // Load chunks within view distance
   294→                for (int dx = -viewDistance; dx <= viewDistance; dx++) {
   295→                        for (int dz = -viewDistance; dz <= viewDistance; dz++) {
   296→                                if (dx * dx + dz * dz > viewDistance * viewDistance) continue;
   297→                                long key = chunkKey(pcx + dx, pcz + dz);
   298→                                if (!chunks.containsKey(key)) {
   299→                                        pendingChunkSends.add(new long[]{pcx + dx, pcz + dz});
   300→                                }
   301→                        }
   302→                }
   303→
   304→                // Unload distant chunks
   305→                java.util.Iterator<java.util.Map.Entry<Long, ChunkData>> it = chunks.entrySet().iterator();
   306→                while (it.hasNext()) {
   307→                        ChunkData chunk = it.next().getValue();
   308→                        int dx = chunk.x - pcx;
   309→                        int dz = chunk.z - pcz;
   310→                        if (dx * dx + dz * dz > (viewDistance + 2) * (viewDistance + 2)) {
   311→                                if (chunk.modified) {
   312→                                        saveChunk(chunk);
   313→                                }
   314→                                it.remove();
   315→                        }
   316→                }
   317→        }

   319→        private void processPendingChunkSends() {
   320→                int sent = 0;
   321→                while (!pendingChunkSends.isEmpty() && sent < 4) {
   322→                        long[] coords = pendingChunkSends.remove(0);
   323→                        int cx = (int) coords[0];
   324→                        int cz = (int) coords[1];
   325→                        ChunkData chunk = getOrGenerateChunk(cx, cz);
   326→                        if (chunk != null) {
   327→                                sendChunkToClient(chunk);
   328→                                sent++;
   329→                        }
   330→                }
   331→        }

   335→        private long chunkKey(int x, int z) {
   336→                return ((long) x & 0xFFFFFFFFL) | (((long) z & 0xFFFFFFFFL) << 32);

   339→        /**
   340→         * Gets or generates a chunk at the given coordinates.
   341→         */
   342→        private ChunkData getOrGenerateChunk(int cx, int cz) {
   343→                long key = chunkKey(cx, cz);
   344→                ChunkData chunk = chunks.get(key);
   345→                if (chunk == null) {
   346→                        chunk = generateChunk(cx, cz);
   347→                        chunks.put(key, chunk);
   348→                }
   349→                if (!chunk.populated) {
   350→                        populateChunk(chunk);
   351→                        chunk.populated = true;
   352→                }
   353→                return chunk;
   354→        }

   356→        /**
   357→         * Generates terrain for a chunk using the world seed.
   358→         */
   359→        private ChunkData generateChunk(int cx, int cz) {
   360→                ChunkData chunk = new ChunkData(cx, cz);
   361→
   362→                // Try loading from IndexedDB first
   363→                byte[] savedData = loadChunkFromDB(cx, cz);
   364→                if (savedData != null) {
   365→                        deserializeChunk(chunk, savedData);
   366→                        chunk.populated = true;
   367→                        return chunk;
   368→                }
   369→
   370→                // Generate terrain
   371→                for (int bx = 0; bx < CHUNK_SIZE; bx++) {
   372→                        for (int bz = 0; bz < CHUNK_SIZE; bz++) {
   373→                                int worldX = cx * CHUNK_SIZE + bx;
   374→                                int worldZ = cz * CHUNK_SIZE + bz;
   375→
   376→                                // Simple height map using seed-based noise
   377→                                int height = generateHeight(worldX, worldZ);
   378→
   379→                                for (int by = 0; by < WORLD_HEIGHT; by++) {
   380→                                        int blockID = 0; // Air
   381→                                        if (by == 0) {
   382→                                                blockID = 7; // Bedrock
   383→                                        } else if (by < height - 4) {
   384→                                                blockID = 1; // Stone
   385→                                        } else if (by < height) {
   386→                                                blockID = 3; // Dirt
   387→                                        } else if (by == height) {
   388→                                                if (height < SEA_LEVEL) {
   389→                                                        blockID = 12; // Sand
   390→                                                } else {
   391→                                                        blockID = 2; // Grass
   392→                                                }
   393→                                        } else if (by <= SEA_LEVEL && by > height) {
   394→                                                blockID = 9; // Water
   395→                                        }
   396→                                        chunk.setBlockID(bx, by, bz, blockID);
   397→                                }
   398→                        }
   399→                }
   400→
   401→                chunk.modified = true;
   402→                return chunk;
   403→        }

   405→        /**
   406→         * Simple height generation using hash-based noise.
   407→         */
   408→        private int generateHeight(int x, int z) {
   409→                long h = worldSeed ^ ((long) x * 341873128712L + (long) z * 132897987541L);
   410→                h = (h ^ (h >>> 30)) * 0xBF58476D1CE4E5B9L;
   411→                h = (h ^ (h >>> 27)) * 0x94D049BB133111EBL;
   412→                h = h ^ (h >>> 31);
   413→                int baseHeight = SEA_LEVEL - 1;
   414→                int variation = (int) ((h & 0xFF) % 20) - 5;
   415→                return baseHeight + variation;
   416→        }

   447→        public void handleBlockChange(byte[] payload) {
   448→                if (payload.length < 13) return;
   449→                int x = readInt(payload, 0);
   450→                int y = readInt(payload, 4);
   451→                int z = readInt(payload, 8);
   452→                int blockID = payload[12] & 0xFF;
   453→
   454→                int cx = x >> 4;
   455→                int cz = z >> 4;
   456→                ChunkData chunk = chunks.get(chunkKey(cx, cz));
   457→                if (chunk != null) {
   458→                        int bx = x & 0xF;
   459→                        int bz = z & 0xF;
   460→                        chunk.setBlockID(bx, y, bz, blockID);
   461→                        // Send block change confirmation to client
   462→                        byte[] response = new byte[13];
   463→                        writeInt(response, 0, x);
   464→                        writeInt(response, 4, y);
   465→                        writeInt(response, 8, z);
   466→                        response[12] = (byte) blockID;
   467→                        WorkerMain.sendIPCMessage(WorkerMain.IPC_CHANNEL_RENDER_COMMAND, response);
   468→                }
   469→        }

   516→        // ========== Serialization ==========
   517→
   518→        private void sendChunkToClient(ChunkData chunk) {
   519→                // Serialize chunk: [4B chunkX][4B chunkZ][blockIDs][metadata]
   520→                int headerSize = 8;
   521→                int blockDataSize = CHUNK_SIZE * CHUNK_SIZE * WORLD_HEIGHT;
   522→                int totalSize = headerSize + blockDataSize + chunk.blockMetadata.length;
   523→                byte[] data = new byte[totalSize];
   524→                writeInt(data, 0, chunk.x);
   525→                writeInt(data, 4, chunk.z);
   526→                System.arraycopy(chunk.blockIDs, 0, data, headerSize, blockDataSize);
   527→                System.arraycopy(chunk.blockMetadata, 0, data, headerSize + blockDataSize, chunk.blockMetadata.length);
   528→                WorkerMain.sendIPCMessage(WorkerMain.IPC_CHANNEL_CHUNK_DATA, data);
   529→        }

   531→        private void deserializeChunk(ChunkData chunk, byte[] data) {
   532→                int blockDataSize = CHUNK_SIZE * CHUNK_SIZE * WORLD_HEIGHT;
   533→                if (data.length >= blockDataSize) {
   534→                        System.arraycopy(data, 0, chunk.blockIDs, 0, blockDataSize);
   535→                        chunk.modified = false;
   536→                }
   537→        }

   545→                chunk.modified = false;

   594→        private static int readInt(byte[] b, int off) {
   595→                return (b[off] << 24) | ((b[off+1] & 0xFF) << 16) | ((b[off+2] & 0xFF) << 8) | (b[off+3] & 0xFF);
   596→        }
   597→
   598→        private static void writeInt(byte[] b, int off, int v) {
   599→                b[off] = (byte)(v >> 24); b[off+1] = (byte)(v >> 16);
   600→                b[off+2] = (byte)(v >> 8); b[off+3] = (byte) v;
