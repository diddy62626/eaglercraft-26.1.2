   161→   161→        private static void registerKeyListeners() {
   162→   162→                canvas.addEventListener("keydown", (EventListener<KeyboardEvent>) event -> {
   163→   163→                        if (imeComposing) return;
   164→   164→                        event.preventDefault();
   165→   165→
   166→   166→                        String code = event.getCode();
   167→   167→                        int keyCode = mapKeyCode(code);
   168→   168→
   169→   169→                        if (keyCode >= 0 && keyCode < MAX_KEYCODE) {
   170→   170→                                if (!keyDown[keyCode]) {
   171→   171→                                        keyDownOnce[keyCode] = true;
   172→   172→                                }
   173→   173→                                keyDown[keyCode] = true;
   174→   174→                        }
   175→   175→
   176→   176→                        currentKeyStr = event.getKey();
   177→   177→                        currentKeyCode = code;
   178→   178→
   179→   179→                        enqueueKeyEvent(keyCode, event.getKey(), code);
   180→   180→                });
   181→   181→
   182→   182→                canvas.addEventListener("keyup", (EventListener<KeyboardEvent>) event -> {
   183→   183→                        if (imeComposing) return;
   184→   184→                        event.preventDefault();
   185→   185→
   186→   186→                        String code = event.getCode();
   187→   187→                        int keyCode = mapKeyCode(code);
   188→   188→
   189→   189→                        if (keyCode >= 0 && keyCode < MAX_KEYCODE) {
   190→   190→                                keyDown[keyCode] = false;
   191→   191→                        }
   192→   192→
   193→   193→                        currentKeyStr = event.getKey();
   194→   194→                        currentKeyCode = code;
   195→   195→                });
   196→   196→
   197→   197→                // IME composition events
   198→   198→                canvas.addEventListener("compositionstart", (EventListener<Event>) event -> {
   199→   199→                        imeComposing = true;
   200→   200→                        compositionText = "";
   201→   201→                });
   202→   202→
   203→   203→                canvas.addEventListener("compositionupdate", (EventListener<Event>) event -> {
   204→   204→                        compositionText = getCompositionData(event);
   205→   205→                });
   206→   206→
   207→   207→                canvas.addEventListener("compositionend", (EventListener<Event>) event -> {
   208→   208→                        imeComposing = false;

   213→   213→                });
   214→   214→        }

   318→   318→                windowAddGamepadConnectedListener((event) -> {
   319→   319→                        int index = getGamepadEventIndex(event);

   330→   330→                                activeGamepadIndex = -1;
   331→   331→                                gamepadConnected = false;
   332→   332→                                // Clear state
   333→   333→                                for (int i = 0; i < gamepadAxes.length; i++) gamepadAxes[i] = 0;
   334→   334→                                for (int i = 0; i < gamepadButtons.length; i++) gamepadButtons[i] = false;
   335→   335→                        }
   336→   336→                        ClientMain.log("[PlatformInput] Gamepad disconnected: index=" + index);
   337→   337→                });
   338→   338→        }
