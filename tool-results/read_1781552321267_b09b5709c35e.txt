   124→   124→         * Sets up the audio processing graph:
   125→   125→         * masterGain <- musicGain <- compressor <- destination
   126→   126→         * masterGain <- sfxGain   <- compressor <- destination
   127→   127→         */
   128→   128→        private static void setupAudioGraph() {
   129→   129→                // Create compressor for loudness normalization
   130→   130→                compressorNode = createDynamicsCompressor(audioContext);
   131→   131→                if (compressorNode != null) {
   132→   132→                        setCompressorThreshold(compressorNode, -24);
   133→   133→                        setCompressorKnee(compressorNode, 30);
   134→   134→                        setCompressorRatio(compressorNode, 12);
   135→   135→                        setCompressorAttack(compressorNode, 0.003);
   136→   136→                        setCompressorRelease(compressorNode, 0.25);
   137→   137→                }
   138→   138→
   139→   139→                // Create gain nodes
   140→   140→                masterGain = createGainNode(audioContext);
   141→   141→                musicGain = createGainNode(audioContext);
   142→   142→                sfxGain = createGainNode(audioContext);
   143→   143→
   144→   144→                // Set initial volumes
   145→   145→                setGainValue(masterGain, masterVolume);
   146→   146→                setGainValue(musicGain, musicVolume);
   147→   147→                setGainValue(sfxGain, sfxVolume);
   148→   148→
   149→   149→                // Connect: musicGain -> masterGain -> compressor -> destination
   150→   150→                connectNodes(musicGain, masterGain);
   151→   151→                connectNodes(sfxGain, masterGain);
   152→   152→
   153→   153→                JSObject destination = getDestination(audioContext);
   154→   154→                if (compressorNode != null) {
   155→   155→                        connectNodes(masterGain, compressorNode);
   156→   156→                        connectNodes(compressorNode, destination);
   157→   157→                } else {
   158→   158→                        connectNodes(masterGain, destination);
   159→   159→                }
   160→   160→        }

   235→   235→        /**
   236→   236→         * Creates an AudioBuffer from PCM float data.
   237→   237→         *
   238→   238→         * @param pcm Float PCM data (mono or interleaved stereo)
   239→   239→         * @param channels Number of channels (1 or 2)
   240→   240→         * @param sr Sample rate of the data
   241→   241→         * @return AudioBuffer JSObject, or null on failure
   242→   242→         */
   243→   243→        public static JSObject createAudioBuffer(float[] pcm, int channels, int sr) {
   244→   244→                if (audioContext == null) return null;
   245→   245→
   246→   246→                int frames = pcm.length / channels;
   247→   247→                JSObject buffer = createAudioBuffer0(audioContext, channels, frames, sr);
   248→   248→                if (buffer == null) return null;
   249→   249→
   250→   250→                // Write channel data
   251→   251→                Float32Array channelData = createFloat32Array(frames);
   252→   252→                for (int ch = 0; ch < channels; ch++) {
   253→   253→                        for (int i = 0; i < frames; i++) {
   254→   254→                                channelData.set(i, pcm[i * channels + ch]);
   255→   255→                        }
   256→   256→                        writeAudioBufferChannel(buffer, ch, channelData);
   257→   257→                }
   258→   258→
   259→   259→                return buffer;
   260→   260→        }
   261→   261→
   262→   262→        /**
   263→   263→         * Creates an AudioBuffer from 16-bit PCM data.
   264→   264→         */
   265→   265→        public static JSObject createAudioBuffer16(Int16Array pcm, int channels, int sr) {
   266→   266→                if (audioContext == null) return null;
   267→   267→
   268→   268→                int frames = pcm.getLength() / channels;
   269→   269→                JSObject buffer = createAudioBuffer0(audioContext, channels, frames, sr);
   270→   270→                if (buffer == null) return null;
   271→   271→
   272→   272→                for (int ch = 0; ch < channels; ch++) {
   273→   273→                        Float32Array channelData = createFloat32Array(frames);
   274→   274→                        for (int i = 0; i < frames; i++) {
   275→   275→                                short sample = (short) pcm.get(i * channels + ch);
   276→   276→                                channelData.set(i, sample / 32768.0f);
   277→   277→                        }
   278→   278→                        writeAudioBufferChannel(buffer, ch, channelData);
   279→   279→                }
   280→   280→
   281→   281→                return buffer;
   282→   282→        }

   297→   297→        /**
   298→   298→         * Plays a sound buffer.
   299→   299→         *
   300→   300→         * @param buffer The AudioBuffer to play
   301→   301→         * @param pitch Playback pitch (1.0 = normal)
   302→   302→         * @param volume Volume (0.0 to 1.0)
   303→   303→         * @param type Sound type (SFX, MUSIC, or VOICE)
   304→   304→         * @return Source node JSObject, or null on failure
   305→   305→         */
   306→   306→        public static JSObject playSound(JSObject buffer, float pitch, float volume, int type) {
   307→   307→                if (audioContext == null || buffer == null) return null;
   308→   308→
   309→   309→                JSObject source = createBufferSource(audioContext);
   310→   310→                if (source == null) return null;
   311→   311→
   312→   312→                setSourceBuffer(source, buffer);
   313→   313→                setSourcePlaybackRate(source, pitch);
   314→   314→                setSourceLoop(source, false);
   315→   315→
   316→   316→                // Create per-source gain for volume control
   317→   317→                JSObject gain = createGainNode(audioContext);
   318→   318→                setGainValue(gain, volume);
   319→   319→
   320→   320→                // Connect: source -> gain -> type-specific gain
   321→   321→                connectNodes(source, gain);
   322→   322→
   323→   323→                switch (type) {
   324→   324→                        case SOUND_TYPE_MUSIC:
   325→   325→                                connectNodes(gain, musicGain);
   326→   326→                                break;
   327→   327→                        case SOUND_TYPE_VOICE:
   328→   328→                        case SOUND_TYPE_SFX:
   329→   329→                        default:
   330→   330→                                connectNodes(gain, sfxGain);
   331→   331→                                break;
   332→   332→                }
   333→   333→
   334→   334→                startSource(source, 0);
   335→   335→                trackSource(source, gain);
   336→   336→
   337→   337→                return source;
   338→   338→        }

   457→   457→        private static void trackSource(JSObject source, JSObject gainNode) {
   458→   458→                if (activeSourceCount < MAX_SOURCES) {
   459→   459→                        activeSources[activeSourceCount++] = source;
   460→   460→                }
   461→   461→
   462→   462→                setSourceOnEnded(source, () -> {
   463→   463→                        for (int i = 0; i < activeSourceCount; i++) {
   464→   464→                                if (activeSources[i] == source) {
   465→   465→                                        activeSources[i] = activeSources[--activeSourceCount];
   466→   466→                                        activeSources[activeSourceCount] = null;
   467→   467→                                        break;
   468→   468→                                }
   469→   469→                        }
   470→   470→                });
   471→   471→        }
