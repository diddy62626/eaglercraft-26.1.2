    13→/**
    14→ * Platform OpenGL implementation for EaglerCraft 26.1.2 on the TeaVM/browser platform.
    15→ * Provides WebGL2-only rendering with no WebGL1 fallback.
    16→ *
    17→ * <p>Key 26.1.2 features:</p>
    18→ * <ul>
    19→ *   <li>WebGL2 native - all GL calls go through WebGL2RenderingContext</li>
    20→ *   <li>EXT_color_buffer_float for floating-point render targets</li>
    21→ *   <li>OES_texture_float_linear for float texture filtering</li>
    22→ *   <li>WEBGL_multi_draw for batched rendering</li>
    23→ *   <li>Transform feedback for GPU particles</li>
    24→ *   <li>Uniform buffer objects for efficient uniform updates</li>
    25→ *   <li>Sampler objects for independent sampler state</li>
    26→ *   <li>Query objects for occlusion and timer queries</li>
    27→ *   <li>3D textures and 2D array textures</li>
    28→ *   <li>Pixel pack/unpack buffer objects for async readback</li>
    29→ *   <li>Vertex array objects (VAO) for efficient state switching</li>
    30→ * </ul>
    31→ */
    32→public class PlatformOpenGL {

    61→        /**
    62→         * Initializes the OpenGL platform. Called by ClientMain during startup.
    63→         * Loads extensions and caches GL capability strings.
    64→         */
    65→        public static void _init() {
    66→                gl = ClientMain.getWebGL2();
    67→                if (gl == null) {
    68→                        throw new RuntimeException("WebGL2 context is null during PlatformOpenGL init!");
    69→                }
    70→
    71→                // Load important extensions
    72→                extColorBufferFloat = gl.getExtension("EXT_color_buffer_float");
    73→                extTextureFloatLinear = gl.getExtension("OES_texture_float_linear");
    74→                extMultiDraw = gl.getExtension("WEBGL_multi_draw");
    75→
    76→                if (extColorBufferFloat == null) {
    77→                        ClientMain.warn("EXT_color_buffer_float not available - float render targets may not work");
    78→                }
    79→                if (extTextureFloatLinear == null) {
    80→                        ClientMain.warn("OES_texture_float_linear not available - float texture filtering may not work");
    81→                }
    82→
    83→                // Cache GL strings
    84→                glVersion = gl.getString(WebGL2RenderingContext.VERSION);
    85→                glVendor = gl.getString(WebGL2RenderingContext.VENDOR);
    86→                glRenderer = gl.getString(WebGL2RenderingContext.RENDERER);
    87→
    88→                ClientMain.log("[PlatformOpenGL] WebGL2 initialized: " + glVersion);
    89→                ClientMain.log("[PlatformOpenGL] Vendor: " + glVendor + ", Renderer: " + glRenderer);
    90→        }

   552→        /**
   553→         * Multi-draw: draws multiple sets of primitives in one call.
   554→         * Uses WEBGL_multi_draw extension if available.
   555→         */
   556→        public static void _wglMultiDrawArrays(int mode, Int32Array firsts, Int32Array counts, int drawcount) {
   557→                if (extMultiDraw != null) {
   558→                        multiDrawArraysEXT(mode, firsts, 0, counts, 0, drawcount);
   559→                } else {
   560→                        // Fallback: loop through draws
   561→                        for (int i = 0; i < drawcount; i++) {
   562→                                gl.drawArrays(mode, firsts.get(i), counts.get(i));
   563→                        }
   564→                }
   565→        }
   566→
   567→        /**
   568→         * Multi-draw elements: draws multiple indexed primitive sets in one call.
   569→         */
   570→        public static void _wglMultiDrawElements(int mode, Int32Array counts, int type, Int32Array offsets, int drawcount) {
   571→                if (extMultiDraw != null) {
   572→                        multiDrawElementsEXT(mode, counts, 0, type, offsets, 0, drawcount);
   573→                } else {
   574→                        for (int i = 0; i < drawcount; i++) {
   575→                                gl.drawElements(mode, counts.get(i), type, offsets.get(i));
   576→                        }
   577→                }
   578→        }
