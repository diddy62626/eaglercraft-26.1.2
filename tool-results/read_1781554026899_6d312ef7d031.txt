   161→        private static void registerKeyListeners() {
   162→                canvas.addEventListener("keydown", (EventListener<KeyboardEvent>) event -> {
   163→                        if (imeComposing) return;
   164→                        event.preventDefault();
   165→
   166→                        String code = event.getCode();
   167→                        int keyCode = mapKeyCode(code);
   168→
   169→                        if (keyCode >= 0 && keyCode < MAX_KEYCODE) {
   170→                                if (!keyDown[keyCode]) {
   171→                                        keyDownOnce[keyCode] = true;
   172→                                }
   173→                                keyDown[keyCode] = true;
   174→                        }
   175→
   176→                        currentKeyStr = event.getKey();
   177→                        currentKeyCode = code;
   178→
   179→                        enqueueKeyEvent(keyCode, event.getKey(), code);
   180→                });
   181→
   182→                canvas.addEventListener("keyup", (EventListener<KeyboardEvent>) event -> {
   183→                        if (imeComposing) return;
   184→                        event.preventDefault();
   185→
   186→                        String code = event.getCode();
   187→                        int keyCode = mapKeyCode(code);
   188→
   189→                        if (keyCode >= 0 && keyCode < MAX_KEYCODE) {
   190→                                keyDown[keyCode] = false;
   191→                        }
   192→
   193→                        currentKeyStr = event.getKey();
   194→                        currentKeyCode = code;
   195→                });
   196→
   197→                // IME composition events
   198→                canvas.addEventListener("compositionstart", (EventListener<Event>) event -> {
   199→                        imeComposing = true;
   200→                        compositionText = "";
   201→                });
   202→
   203→                canvas.addEventListener("compositionupdate", (EventListener<Event>) event -> {
   204→                        compositionText = getCompositionData(event);
   205→                });
   206→
   207→                canvas.addEventListener("compositionend", (EventListener<Event>) event -> {
   208→                        imeComposing = false;
   209→                        compositionText = getCompositionData(event);
   210→                        // Insert the final composition text as if it were typed
   211→                        currentKeyStr = compositionText;
   212→                        currentKeyCode = "";
   213→                });
   214→        }

   216→        private static void registerMouseListeners() {
   217→                canvas.addEventListener("mousedown", (EventListener<MouseEvent>) event -> {
   218→                        event.preventDefault();
   219→                        if (pointerLocked) return;
   220→
   221→                        int button = event.getButton();
   222→                        if (button >= 0 && button < MAX_MOUSEBUTTON) {
   223→                                mouseDown[button] = true;
   224→                        }
   225→                });
   226→
   227→                canvas.addEventListener("mouseup", (EventListener<MouseEvent>) event -> {
   228→                        event.preventDefault();
   229→                        if (pointerLocked) return;
   230→
   231→                        int button = event.getButton();
   232→                        if (button >= 0 && button < MAX_MOUSEBUTTON) {
   233→                                mouseDown[button] = false;
   234→                        }
   235→                });
   236→
   237→                canvas.addEventListener("mousemove", (EventListener<MouseEvent>) event -> {
   238→                        if (pointerLocked) {
   239→                                // In pointer lock mode, use movementX/Y for delta
   240→                                dx += getMovementX(event);
   241→                                dy += getMovementY(event);
   242→                        } else {
   243→                                // Normal mode: track absolute position relative to canvas
   244→                                int[] coords = getCanvasRelativeCoords(event);
   245→                                mouseX = coords[0];
   246→                                mouseY = coords[1];
   247→                        }
   248→                });
   249→        }

   315→        private static void registerGamepadListeners() {
   316→                if (!ClientMain.gamepadSupported) return;
   317→
   318→                windowAddGamepadConnectedListener((event) -> {
   319→                        int index = getGamepadEventIndex(event);
   320→                        if (activeGamepadIndex < 0) {
   321→                                activeGamepadIndex = index;
   322→                        }
   323→                        gamepadConnected = true;
   324→                        ClientMain.log("[PlatformInput] Gamepad connected: index=" + index);
   325→                });
   326→
   327→                windowAddGamepadDisconnectedListener((event) -> {
   328→                        int index = getGamepadEventIndex(event);
   329→                        if (index == activeGamepadIndex) {
   330→                                activeGamepadIndex = -1;
   331→                                gamepadConnected = false;
   332→                                // Clear state
   333→                                for (int i = 0; i < gamepadAxes.length; i++) gamepadAxes[i] = 0;
   334→                                for (int i = 0; i < gamepadButtons.length; i++) gamepadButtons[i] = false;
   335→                        }
   336→                        ClientMain.log("[PlatformInput] Gamepad disconnected: index=" + index);
   337→                });
   338→        }
