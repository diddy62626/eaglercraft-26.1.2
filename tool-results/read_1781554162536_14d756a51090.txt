    12→/**
    13→ * Platform audio implementation for EaglerCraft 26.1.2 on the TeaVM/browser platform.
    14→ * Uses the Web Audio API for sound playback, with AudioWorklet for custom processing.
    15→ *
    16→ * <p>Key 26.1.2 improvements:</p>
    17→ * <ul>
    18→ *   <li>AudioContext with configurable sample rate</li>
    19→ *   <li>AudioWorklet for custom audio processing pipelines</li>
    20→ *   <li>DynamicsCompressorNode for loudness normalization</li>
    21→ *   <li>Spatial audio via PannerNode for 3D positional sound</li>
    22→ *   <li>Efficient buffer pooling to reduce GC pressure</li>
    23→ *   <li>Music crossfade with GainNode ramping</li>
    24→ *   <li>Vorbis decoding (via ogg-opus-decoder or WebCodecs)</li>
    25→ *   <li>Master volume control with smooth ramping</li>
    26→ *   <li>Channel splitting for separate music/SFX volume</li>
    27→ * </ul>

    91→                        return;
    92→                }

    98→                }

   101→                if (audioContext == null) {
   102→                        ClientMain.warn("[PlatformAudio] Failed to create AudioContext");
   103→                        return;
   104→                }
   105→
   106→                // Check AudioWorklet availability
   107→                audioWorkletAvailable = checkAudioWorklet();
   108→
   109→                // Get actual sample rate
   110→                sampleRate = getAudioContextSampleRate(audioContext);
   111→
   112→                // Store in JS global for native method access
   113→                setAudioContextGlobal(audioContext);
   114→
   115→                // Create audio processing graph
   116→                setupAudioGraph();
   117→
   118→                initialized = true;
   119→                ClientMain.log("[PlatformAudio] Audio initialized (sampleRate=" + sampleRate
   120→                                + ", worklet=" + audioWorkletAvailable + ")");
   121→        }

   123→        /**
   124→         * Sets up the audio processing graph:
   125→         * masterGain <- musicGain <- compressor <- destination
   126→         * masterGain <- sfxGain   <- compressor <- destination
   127→         */
   128→        private static void setupAudioGraph() {
   129→                // Create compressor for loudness normalization
   130→                compressorNode = createDynamicsCompressor(audioContext);
   131→                if (compressorNode != null) {
   132→                        setCompressorThreshold(compressorNode, -24);
   133→                        setCompressorKnee(compressorNode, 30);
   134→                        setCompressorRatio(compressorNode, 12);
   135→                        setCompressorAttack(compressorNode, 0.003);
   136→                        setCompressorRelease(compressorNode, 0.25);
   137→                }
   138→
   139→                // Create gain nodes
   140→                masterGain = createGainNode(audioContext);
   141→                musicGain = createGainNode(audioContext);
   142→                sfxGain = createGainNode(audioContext);
   143→
   144→                // Set initial volumes
   145→                setGainValue(masterGain, masterVolume);
   146→                setGainValue(musicGain, musicVolume);
   147→                setGainValue(sfxGain, sfxVolume);
   148→
   149→                // Connect: musicGain -> masterGain -> compressor -> destination
   150→                connectNodes(musicGain, masterGain);
   151→                connectNodes(sfxGain, masterGain);
   152→
   153→                JSObject destination = getDestination(audioContext);
   154→                if (compressorNode != null) {
   155→                        connectNodes(masterGain, compressorNode);
   156→                        connectNodes(compressorNode, destination);
   157→                } else {
   158→                        connectNodes(masterGain, destination);
   159→                }
   160→        }

   235→        /**
   236→         * Creates an AudioBuffer from PCM float data.
   237→         *
   238→         * @param pcm Float PCM data (mono or interleaved stereo)
   239→         * @param channels Number of channels (1 or 2)
   240→         * @param sr Sample rate of the data
   241→         * @return AudioBuffer JSObject, or null on failure
   242→         */
   243→        public static JSObject createAudioBuffer(float[] pcm, int channels, int sr) {
   244→                if (audioContext == null) return null;
   245→
   246→                int frames = pcm.length / channels;
   247→                JSObject buffer = createAudioBuffer0(audioContext, channels, frames, sr);
   248→                if (buffer == null) return null;
   249→
   250→                // Write channel data
   251→                Float32Array channelData = createFloat32Array(frames);
   252→                for (int ch = 0; ch < channels; ch++) {
   253→                        for (int i = 0; i < frames; i++) {
   254→                                channelData.set(i, pcm[i * channels + ch]);
   255→                        }
   256→                        writeAudioBufferChannel(buffer, ch, channelData);
   257→                }
   258→
   259→                return buffer;
   260→        }
   261→
   262→        /**
   263→         * Creates an AudioBuffer from 16-bit PCM data.
   264→         */
   265→        public static JSObject createAudioBuffer16(Int16Array pcm, int channels, int sr) {
   266→                if (audioContext == null) return null;
   267→
   268→                int frames = pcm.getLength() / channels;
   269→                JSObject buffer = createAudioBuffer0(audioContext, channels, frames, sr);
   270→                if (buffer == null) return null;
   271→
   272→                for (int ch = 0; ch < channels; ch++) {
   273→                        Float32Array channelData = createFloat32Array(frames);
   274→                        for (int i = 0; i < frames; i++) {
   275→                                short sample = (short) pcm.get(i * channels + ch);
   276→                                channelData.set(i, sample / 32768.0f);
   277→                        }
   278→                        writeAudioBufferChannel(buffer, ch, channelData);
   279→                }
   280→
   281→                return buffer;
   282→        }

   295→        // ========== Sound Playback ==========
   296→
   297→        /**
   298→         * Plays a sound buffer.
   299→         *
   300→         * @param buffer The AudioBuffer to play
   301→         * @param pitch Playback pitch (1.0 = normal)
   302→         * @param volume Volume (0.0 to 1.0)
   303→         * @param type Sound type (SFX, MUSIC, or VOICE)
   304→         * @return Source node JSObject, or null on failure
   305→         */
   306→        public static JSObject playSound(JSObject buffer, float pitch, float volume, int type) {
   307→                if (audioContext == null || buffer == null) return null;
   308→
   309→                JSObject source = createBufferSource(audioContext);
   310→                if (source == null) return null;
   311→
   312→                setSourceBuffer(source, buffer);
   313→                setSourcePlaybackRate(source, pitch);
   314→                setSourceLoop(source, false);
   315→
   316→                // Create per-source gain for volume control
   317→                JSObject gain = createGainNode(audioContext);
   318→                setGainValue(gain, volume);
   319→
   320→                // Connect: source -> gain -> type-specific gain
   321→                connectNodes(source, gain);
   322→
   323→                switch (type) {
   324→                        case SOUND_TYPE_MUSIC:
   325→                                connectNodes(gain, musicGain);
   326→                                break;
   327→                        case SOUND_TYPE_VOICE:
   328→                        case SOUND_TYPE_SFX:
   329→                        default:
   330→                                connectNodes(gain, sfxGain);
   331→                                break;
   332→                }
   333→
   334→                startSource(source, 0);
   335→                trackSource(source, gain);
   336→
   337→                return source;
   338→        }

   340→        /**
   341→         * Plays a looping sound (for music).
   342→         */
   343→        public static JSObject playMusic(JSObject buffer, float volume, float fadeIn) {
   344→                if (audioContext == null || buffer == null) return null;
   345→
   346→                JSObject source = createBufferSource(audioContext);
   347→                if (source == null) return null;
   348→
   349→                setSourceBuffer(source, buffer);
   350→                setSourceLoop(source, true);
   351→                setSourcePlaybackRate(source, 1.0f);
   352→
   353→                JSObject gain = createGainNode(audioContext);
   354→                if (fadeIn > 0) {
   355→                        setGainValue(gain, 0.0f);
   356→                        rampGainValue(gain, volume, fadeIn);
   357→                } else {
   358→                        setGainValue(gain, volume);
   359→                }
   360→
   361→                connectNodes(source, gain);
   362→                connectNodes(gain, musicGain);
   363→
   364→                startSource(source, 0);
   365→                trackSource(source, gain);
   366→
   367→                return source;
   368→        }

   383→        /**
   384→         * Plays a 3D positional sound.
   385→         */
   386→        public static JSObject playPositionedSound(JSObject buffer, float x, float y, float z,
   387→                                                                                                float pitch, float volume) {
   388→                if (audioContext == null || buffer == null) return null;
   389→
   390→                JSObject source = createBufferSource(audioContext);
   391→                if (source == null) return null;
   392→
   393→                setSourceBuffer(source, buffer);
   394→                setSourcePlaybackRate(source, pitch);
   395→                setSourceLoop(source, false);
   396→
   397→                JSObject panner = createPanner(audioContext);
   398→                if (panner != null) {
   399→                        setPannerPosition(panner, x, y, z);
   400→                        setPannerDistanceModel(panner, "inverse");
   401→                        setPannerRefDistance(panner, 1.0f);
   402→                        setPannerMaxDistance(panner, 64.0f);
   403→                        setPannerRolloffFactor(panner, 1.0f);
   404→
   405→                        JSObject gain = createGainNode(audioContext);
   406→                        setGainValue(gain, volume);
   407→
   408→                        connectNodes(source, panner);
   409→                        connectNodes(panner, gain);
   410→                        connectNodes(gain, sfxGain);
   411→                } else {
   412→                        JSObject gain = createGainNode(audioContext);
   413→                        setGainValue(gain, volume);
   414→                        connectNodes(source, gain);
   415→                        connectNodes(gain, sfxGain);
   416→                }
   417→
   418→                startSource(source, 0);
   419→                trackSource(source, null);
   420→
   421→                return source;
   422→        }

   455→        // ========== Source Tracking ==========
   456→
   457→        private static void trackSource(JSObject source, JSObject gainNode) {
   458→                if (activeSourceCount < MAX_SOURCES) {
   459→                        activeSources[activeSourceCount++] = source;
   460→                }
   461→
   462→                setSourceOnEnded(source, () -> {
   463→                        for (int i = 0; i < activeSourceCount; i++) {
   464→                                if (activeSources[i] == source) {
   465→                                        activeSources[i] = activeSources[--activeSourceCount];
   466→                                        activeSources[activeSourceCount] = null;
   467→                                        break;
   468→                                }
   469→                        }
   470→                });
   471→        }
   472→
   473→        /**
   474→         * Stops all currently playing sounds.
   475→         */
   476→        public static void stopAllSounds() {
   477→                for (int i = 0; i < activeSourceCount; i++) {
   478→                        if (activeSources[i] != null) {
   479→                                stopSource(activeSources[i]);
   480→                                activeSources[i] = null;
   481→                        }
   482→                }
   483→                activeSourceCount = 0;
   484→        }
