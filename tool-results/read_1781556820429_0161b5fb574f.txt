    14→/**

    16→ *
    17→ * <p>This class bridges the TeaVM platform layer (initialized by
    18→ * {@link ClientMain}) with Minecraft 26.1.2's game code. It is called
    19→ * after the platform subsystems are ready and is responsible for:</p>
    20→ * <ol>
    21→ *   <li>Initializing the EaglerCraft adapter layer (config, profile, services)</li>
    22→ *   <li>Setting up the WebGL2 rendering pipeline for MC's renderer</li>

    24→ *   <li>Handling frame-by-frame tick/render dispatch</li>

    28→ * <h3>Initialization sequence:</h3>
    29→ * <pre>
    30→ *   1. MainClass.main()           → TeaVM entry point

    36→ *      c. PlatformWebService.init() → Verify service availability

   202→        /**
   203→         * Sets up the WebGL2 rendering pipeline for Minecraft's renderer.
   204→         *
   205→         * <p>Configures the initial GL state that MC's GameRenderer expects:</p>
   206→         * <ul>
   207→         *   <li>Set viewport to canvas size</li>
   208→         *   <li>Enable depth testing</li>
   209→         *   <li>Enable back-face culling</li>
   210→         *   <li>Set up the clear color</li>
   211→         *   <li>Configure blending for transparency</li>
   212→         * </ul>
   213→         */
   214→        private static void setupRenderingPipeline() {
   215→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   216→                        ClientMain.log("[EaglerCraft] Setting up rendering pipeline...");
   217→                }
   218→
   219→                // Set initial viewport
   220→                int width = PlatformRuntime.getCanvasDrawableWidth();
   221→                int height = PlatformRuntime.getCanvasDrawableHeight();
   222→                PlatformOpenGL._wglViewport(0, 0, width, height);
   223→
   224→                // Enable depth testing (MC requires this)
   225→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.DEPTH_TEST);
   226→                PlatformOpenGL._wglDepthFunc(WebGL2RenderingContext.LEQUAL);
   227→
   228→                // Enable back-face culling (MC uses this for block faces)
   229→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.CULL_FACE);
   230→                PlatformOpenGL._wglCullFace(WebGL2RenderingContext.BACK);
   231→                PlatformOpenGL._wglFrontFace(WebGL2RenderingContext.CCW);
   232→
   233→                // Set default clear color (sky blue-ish, matches MC default)
   234→                PlatformOpenGL._wglClearColor(0.627f, 0.808f, 1.0f, 1.0f);
   235→
   236→                // Configure blending for transparency (used heavily by MC)
   237→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.BLEND);
   238→                PlatformOpenGL._wglBlendFunc(WebGL2RenderingContext.SRC_ALPHA,
   239→                                WebGL2RenderingContext.ONE_MINUS_SRC_ALPHA);
   240→
   241→                // Disable stencil test by default (MC uses it for some effects)
   242→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.STENCIL_TEST);
   243→
   244→                // Enable scissor test (used by MC for viewport clipping)
   245→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.SCISSOR_TEST);
   246→
   247→                // Reset shader stats
   248→                EaglerShaderImpl.resetStats();
   249→
   250→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   251→                        ClientMain.log("[EaglerCraft] Rendering pipeline configured "
   252→                                        + "(viewport: " + width + "x" + height + ")");
   253→                        ClientMain.log("[EaglerCraft] WebGL2 Version: " + PlatformOpenGL.getGLVersion());
   254→                        ClientMain.log("[EaglerCraft] GL Vendor: " + PlatformOpenGL.getGLVendor());
   255→                        ClientMain.log("[EaglerCraft] GL Renderer: " + PlatformOpenGL.getGLRenderer());
   256→                }
   257→        }

   336→        /**
   337→         * Called once per frame by the ClientMain game loop.
   338→         *
   339→         * <p>This method handles the frame-by-frame update cycle:</p>
   340→         * <ol>
   341→         *   <li>Check if the page is visible (auto-pause when hidden)</li>
   342→         *   <li>Process input events from PlatformInput</li>
   343→         *   <li>Run game logic ticks at 20 TPS (fixed timestep)</li>
   344→         *   <li>Render the frame</li>
   345→         *   <li>Update FPS counter and timing</li>
   346→         * </ol>
   347→         *
   348→         * <p>The tick rate is decoupled from the frame rate: the game logic
   349→         * runs at a fixed 20 TPS, while rendering happens at the display's
   350→         * refresh rate (typically 60Hz via requestAnimationFrame). This
   351→         * prevents simulation instability from variable frame timing.</p>
   352→         *
   353→         * <p>TODO: Once MC 26.1.2 is decompiled, this method should be
   354→         * patched to call the actual MC tick/render methods. Expected:</p>
   355→         * <pre>
   356→         *   Minecraft mc = Minecraft.getInstance();
   357→         *   mc.runTick(isPaused);
   358→         * </pre>
   359→         */
   360→        public static void tick() {
   361→                if (gameState != STATE_RUNNING) return;
   362→
   363→                long currentTime = PlatformRuntime.getCurrentTimeMillis();
   364→                long deltaTime = currentTime - lastTickTime;
   365→
   366→                // Clamp delta time to prevent spiral of death
   367→                if (deltaTime > EaglerCraftConfig.MAX_FRAME_DELTA_MS) {
   368→                        deltaTime = EaglerCraftConfig.MAX_FRAME_DELTA_MS;
   369→                }
   370→
   371→                // Auto-pause when page is not visible
   372→                if (!PlatformRuntime.isPageVisible()) {
   373→                        if (!paused) {
   374→                                paused = true;
   375→                                gameState = STATE_PAUSED;
   376→                        }
   377→                        return;
   378→                } else if (paused) {
   379→                        paused = false;
   380→                        gameState = STATE_RUNNING;
   381→                        lastTickTime = currentTime;
   382→                        tickAccumulator = 0;
   383→                        return;
   384→                }
   385→
   386→                // Process input
   387→                processInput();
   388→
   389→                // Run game logic ticks at fixed TPS
   390→                if (EaglerCraftConfig.USE_FIXED_TIMESTEP) {
   391→                        tickAccumulator += deltaTime;
   392→                        while (tickAccumulator >= EaglerCraftConfig.MS_PER_TICK) {
   393→                                gameLogicTick();
   394→                                tickAccumulator -= EaglerCraftConfig.MS_PER_TICK;
   395→                        }
   396→                } else {
   397→                        // Variable timestep: run one tick per frame
   398→                        // MC internally handles the tick rate, so we just call it
   399→                        gameLogicTick();
   400→                }
   401→
   402→                // Render frame
   403→                renderFrame();
   404→
   405→                // Update timing
   406→                lastTickTime = currentTime;
   407→                totalFrames++;
   408→        }
