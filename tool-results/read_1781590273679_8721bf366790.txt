    11→/**
    12→ * Shader adapter that bridges Minecraft 26.1.2's shader system to WebGL2
    13→ * through {@link PlatformOpenGL}.
    14→ *
    15→ * <p>In vanilla MC 26.1.2, shaders are compiled and managed through LWJGL's
    16→ * OpenGL bindings. In EaglerCraft, all shader operations must go through
    17→ * WebGL2, which has a similar but not identical API. This class provides:</p>
    18→ * <ul>
    19→ *   <li>GLSL ES 3.00 shader source preprocessing (WebGL2 requires #version 300 es)</li>
    20→ *   <li>Shader compilation with error logging and fallback handling</li>
    21→ *   <li>Program linking with reflection for uniforms and attributes</li>
    22→ *   <li>Uniform buffer object (UBO) management for efficient state updates</li>
    23→ *   <li>Shader caching to avoid recompilation on context loss recovery</li>
    24→ * </ul>
    25→ *
    26→ * <h3>GLSL compatibility notes:</h3>
    27→ * <p>MC 26.1.2 desktop shaders typically use GLSL 1.20-1.50 syntax. WebGL2
    28→ * requires GLSL ES 3.00. Key differences handled by this adapter:</p>
    29→ * <ul>
    30→ *   <li><code>attribute</code> → <code>in</code> (vertex shader)</li>
    31→ *   <li><code>varying</code> → <code>out</code> (vertex) / <code>in</code> (fragment)</li>
    32→ *   <li><code>texture2D</code> → <code>texture</code></li>
    33→ *   <li><code>gl_FragColor</code> → explicit output variable declaration</li>
    34→ *   <li>Precision qualifiers required (<code>highp</code>, <code>mediump</code>, <code>lowp</code>)</li>
    35→ *   <li>Integer attributes require <code>flat</code> interpolation qualifier</li>
    36→ * </ul>
    37→ *
    38→ * <p>Architecture note: This class is in the <code>main</code> source set's
    39→ * <code>adapter</code> sub-package. MC 26.1.2's shader classes (e.g.,
    40→ * {@code ShaderProgram}, {@code ShaderInstance}) will be patched to call
    41→ * this adapter instead of LWJGL directly.</p>
    42→ *
    43→ * TODO: Update to actual MC 26.1.2 class names after decompilation.
    44→ * Expected: net.minecraft.client.renderer.ShaderInstance or similar
    45→ */
    46→public class EaglerShaderImpl {

    87→        // ========== Shader Compilation ==========
    88→
    89→        /**
    90→         * Compiles a GLSL shader source string for WebGL2.
    91→         *
    92→         * <p>This method handles the necessary source transformations to convert
    93→         * desktop GLSL to GLSL ES 3.00 compatible source, then compiles it
    94→         * using {@link PlatformOpenGL#_wglCreateShader}, _wglShaderSource,
    95→         * and _wglCompileShader.</p>
    96→         *
    97→         * <p>If compilation fails, the error log is printed and null is returned.</p>
    98→         *
    99→         * @param type   the shader type ({@link #VERTEX_SHADER} or {@link #FRAGMENT_SHADER})
   100→         * @param source the GLSL source code (desktop GLSL is auto-converted)
   101→         * @param name   a descriptive name for logging (e.g., "rendertype_solid.vsh")
   102→         * @return the compiled shader JSObject handle, or null if compilation failed
   103→         */
   104→        public static JSObject compileShader(int type, String source, String name) {
   105→                // Preprocess the source for GLSL ES 3.00 compatibility
   106→                String processedSource = preprocessShaderSource(type, source);
   107→
   108→                // Create and compile the shader
   109→                JSObject shader = PlatformOpenGL._wglCreateShader(type);
   110→                if (shader == null) {
   111→                        ClientMain.error("[EaglerShaderImpl] Failed to create shader object for: " + name);
   112→                        shaderCompileFailures++;
   113→                        return null;
   114→                }
   115→
   116→                PlatformOpenGL._wglShaderSource(shader, processedSource);
   117→                PlatformOpenGL._wglCompileShader(shader);
   118→
   119→                // Check compilation status
   120→                boolean compiled = PlatformOpenGL._wglGetShaderCompiled(shader);
   121→                if (!compiled) {
   122→                        String infoLog = PlatformOpenGL._wglGetShaderInfoLog(shader);
   123→                        ClientMain.error("[EaglerShaderImpl] Shader compilation failed: " + name);
   124→                        ClientMain.error("[EaglerShaderImpl] Info log: " + infoLog);
   125→
   126→                        // Log the first few lines of the processed source for debugging
   127→                        if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   128→                                String[] lines = processedSource.split("\n");
   129→                                int maxLines = Math.min(lines.length, 20);
   130→                                ClientMain.error("[EaglerShaderImpl] Source preview (" + lines.length + " lines total):");
   131→                                for (int i = 0; i < maxLines; i++) {
   132→                                        ClientMain.error("  " + (i + 1) + ": " + lines[i]);
   133→                                }
   134→                                if (lines.length > maxLines) {
   135→                                        ClientMain.error("  ... (" + (lines.length - maxLines) + " more lines)");
   136→                                }
   137→                        }
   138→
   139→                        PlatformOpenGL._wglDeleteShader(shader);
   140→                        shaderCompileFailures++;
   141→                        return null;
   142→                }
   143→
   144→                shadersCompiled++;
   145→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   146→                        ClientMain.log("[EaglerShaderImpl] Compiled shader: " + name);
   147→                }
   148→
   149→                return shader;
   150→        }

   176→        // ========== Program Linking ==========
   177→
   178→        /**
   179→         * Creates and links a shader program from vertex and fragment shaders.
   180→         *
   181→         * <p>After successful linking, the shaders are detached and deleted
   182→         * (they are no longer needed once the program is linked).</p>
   183→         *
   184→         * @param vertexShader   compiled vertex shader handle
   185→         * @param fragmentShader compiled fragment shader handle
   186→         * @param name           descriptive name for logging (e.g., "rendertype_solid")
   187→         * @return the linked program JSObject handle, or null if linking failed
   188→         */
   189→        public static JSObject linkProgram(JSObject vertexShader, JSObject fragmentShader, String name) {
   190→                if (vertexShader == null || fragmentShader == null) {
   191→                        ClientMain.error("[EaglerShaderImpl] Cannot link program with null shader: " + name);
   192→                        programLinkFailures++;
   193→                        return null;
   194→                }
   195→
   196→                JSObject program = PlatformOpenGL._wglCreateProgram();
   197→                if (program == null) {
   198→                        ClientMain.error("[EaglerShaderImpl] Failed to create program object: " + name);
   199→                        programLinkFailures++;
   200→                        return null;
   201→                }
   202→
   203→                PlatformOpenGL._wglAttachShader(program, vertexShader);
   204→                PlatformOpenGL._wglAttachShader(program, fragmentShader);
   205→                PlatformOpenGL._wglLinkProgram(program);
   206→
   207→                // Check link status
   208→                boolean linked = PlatformOpenGL._wglGetProgramLinked(program);
   209→                if (!linked) {
   210→                        String infoLog = PlatformOpenGL._wglGetProgramInfoLog(program);
   211→                        ClientMain.error("[EaglerShaderImpl] Program link failed: " + name);
   212→                        ClientMain.error("[EaglerShaderImpl] Info log: " + infoLog);
   213→
   214→                        PlatformOpenGL._wglDetachShader(program, vertexShader);
   215→                        PlatformOpenGL._wglDetachShader(program, fragmentShader);
   216→                        PlatformOpenGL._wglDeleteProgram(program);
   217→                        programLinkFailures++;
   218→                        return null;
   219→                }
   220→
   221→                // Detach and delete shaders (no longer needed after linking)
   222→                PlatformOpenGL._wglDetachShader(program, vertexShader);
   223→                PlatformOpenGL._wglDetachShader(program, fragmentShader);
   224→                PlatformOpenGL._wglDeleteShader(vertexShader);
   225→                PlatformOpenGL._wglDeleteShader(fragmentShader);
   226→
   227→                programsLinked++;
   228→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   229→                        ClientMain.log("[EaglerShaderImpl] Linked program: " + name);
   230→                }
   231→
   232→                return program;
   233→        }

   235→        /**
   236→         * Creates a complete shader program from source strings.
   237→         * Convenience method that compiles both shaders and links them in one call.
   238→         *
   239→         * @param vertexSource   vertex shader GLSL source
   240→         * @param fragmentSource fragment shader GLSL source
   241→         * @param name           descriptive name for logging
   242→         * @return the linked program handle, or null if any step failed
   243→         */
   244→        public static JSObject createProgramFromSource(String vertexSource, String fragmentSource, String name) {
   245→                JSObject vertexShader = compileVertexShader(vertexSource, name + ".vsh");
   246→                if (vertexShader == null) return null;
   247→
   248→                JSObject fragmentShader = compileFragmentShader(fragmentSource, name + ".fsh");
   249→                if (fragmentShader == null) {
   250→                        PlatformOpenGL._wglDeleteShader(vertexShader);
   251→                        return null;
   252→                }
   253→
   254→                return linkProgram(vertexShader, fragmentShader, name);
   255→        }

   333→        // ========== GLSL Source Preprocessing ==========
   334→
   335→        /**
   336→         * Preprocesses a GLSL source string for WebGL2 / GLSL ES 3.00 compatibility.
   337→         *
   338→         * <p>This method performs the following transformations:</p>
   339→         * <ol>
   340→         *   <li>Strips any existing <code>#version</code> directive</li>
   341→         *   <li>Adds <code>#version 300 es</code> header</li>
   342→         *   <li>Adds default precision qualifiers</li>
   343→         *   <li>Converts desktop GLSL keywords to GLSL ES equivalents:
   344→         *     <ul>
   345→         *       <li><code>attribute</code> → <code>in</code> (vertex shader)</li>
   346→         *       <li><code>varying</code> → <code>out</code> (vertex) / <code>in</code> (fragment)</li>
   347→         *       <li><code>texture2D</code> → <code>texture</code></li>
   348→         *       <li><code>textureCube</code> → <code>texture</code></li>
   349→         *     </ul>
   350→         *   </li>
   351→         *   <li>Handles <code>gl_FragColor</code> → explicit output declaration</li>
   352→         * </ol>
   353→         *
   354→         * <p>Note: This is a best-effort preprocessor. Complex shaders with
   355→         * preprocessor macros, multiple output targets, or advanced GLSL features
   356→         * may require manual adjustment. The EaglerCraft shader patches (applied
   357→         * at build time) handle the full conversion.</p>
   358→         *
   359→         * @param type   the shader type
   360→         * @param source the original GLSL source
   361→         * @return the preprocessed source compatible with GLSL ES 3.00
   362→         */
   363→        public static String preprocessShaderSource(int type, String source) {
   364→                StringBuilder result = new StringBuilder();
   365→
   366→                // Add GLSL ES version header
   367→                result.append(GLSL_ES_HEADER).append("\n");
   368→
   369→                // Add precision qualifiers
   370→                if (type == FRAGMENT_SHADER) {
   371→                        result.append(FRAGMENT_DEFAULT_PRECISION);
   372→                } else {
   373→                        result.append(VERTEX_DEFAULT_PRECISION);
   374→                }
   375→
   376→                // Process each line
   377→                String[] lines = source.split("\n");
   378→                boolean hasFragColor = false;
   379→
   380→                for (String line : lines) {
   381→                        String trimmed = line.trim();
   382→
   383→                        // Skip #version directives (we already added our own)
   384→                        if (trimmed.startsWith("#version")) {
   385→                                continue;
   386→                        }
   387→
   388→                        // Skip #extension directives that are desktop-only
   389→                        if (trimmed.startsWith("#extension GL_ARB_") || trimmed.startsWith("#extension GL_EXT_")) {
   390→                                // Some extensions have WebGL2 equivalents, but most can be safely skipped
   391→                                continue;
   392→                        }
   393→
   394→                        // Convert desktop GLSL keywords to GLSL ES 3.00
   395→                        String converted = line;
   396→
   397→                        if (type == VERTEX_SHADER) {
   398→                                // attribute → in (vertex shader)
   399→                                converted = convertKeyword(converted, "attribute", "in");
   400→                                // varying → out (vertex shader)
   401→                                converted = convertKeyword(converted, "varying", "out");
   402→                        } else {
   403→                                // varying → in (fragment shader)
   404→                                converted = convertKeyword(converted, "varying", "in");
   405→                        }
   406→
   407→                        // texture2D → texture (GLSL ES 3.00 unified texture function)
   408→                        converted = convertTextureCall(converted, "texture2D");
   409→                        converted = convertTextureCall(converted, "textureCube");
   410→
   411→                        // Track gl_FragColor usage for output variable injection
   412→                        if (converted.contains("gl_FragColor")) {
   413→                                hasFragColor = true;
   414→                        }
   415→
   416→                        result.append(converted).append("\n");
   417→                }
   418→
   419→                // If the fragment shader uses gl_FragColor, we need to inject an output
   420→                // variable declaration and replace gl_FragColor references
   421→                if (type == FRAGMENT_SHADER && hasFragColor) {
   422→                        String processed = result.toString();
   423→                        // Replace gl_FragColor with our output variable
   424→                        processed = processed.replace("gl_FragColor", "eagler_FragColor");
   425→                        // Insert the output variable declaration after the version header + precision
   426→                        int insertPos = processed.indexOf('\n', processed.indexOf('\n') + 1) + 1;
   427→                        insertPos = processed.indexOf('\n', insertPos) + 1; // After precision lines
   428→                        processed = processed.substring(0, insertPos)
   429→                                        + "out vec4 eagler_FragColor;\n"
   430→                                        + processed.substring(insertPos);
   431→                        return processed;
   432→                }
   433→
   434→                return result.toString();
   435→        }

   437→        /**
   438→         * Converts a GLSL keyword from desktop to ES 3.00 syntax.
   439→         * Only replaces the keyword when it appears as a standalone token
   440→         * (not as part of a longer identifier).
   441→         *
   442→         * @param line     the source line
   443→         * @param oldWord  the keyword to replace
   444→         * @param newWord  the replacement keyword
   445→         * @return the line with replacements applied
   446→         */
   447→        private static String convertKeyword(String line, String oldWord, String newWord) {
   448→                // Simple word boundary replacement
   449→                // Replace "attribute " with "in " and similar patterns
   450→                return line.replaceAll("\\b" + oldWord + "\\b", newWord);
   451→        }

   467→        /**
   468→         * Creates a uniform buffer object (UBO) for efficient uniform updates.
   469→         * UBOs allow updating multiple uniforms in a single buffer upload,
   470→         * which is significantly faster than individual uniform calls.
   471→         *
   472→         * <p>Usage pattern:</p>
   473→         * <pre>
   474→         *   JSObject ubo = EaglerShaderImpl.createUniformBuffer(matricesData, GL_STATIC_DRAW);
   475→         *   EaglerShaderImpl.bindUniformBufferBase(ubo, 0); // bind to binding point 0
   476→         *   // In shader: layout(std140) uniform MatricesBlock { ... } matricesBlock;
   477→         * </pre>
   478→         *
   479→         * @param data   the initial buffer data as a Float32Array
   480→         * @param usage  the buffer usage hint (GL_STATIC_DRAW, GL_DYNAMIC_DRAW, etc.)
   481→         * @return the buffer JSObject handle
   482→         */
   483→        public static JSObject createUniformBuffer(Float32Array data, int usage) {
   484→                JSObject buffer = PlatformOpenGL._wglCreateBuffer();
   485→                PlatformOpenGL._wglBindBuffer(WebGL2RenderingContext.UNIFORM_BUFFER, buffer);
   486→                PlatformOpenGL._wglBufferData(WebGL2RenderingContext.UNIFORM_BUFFER, data, usage);
   487→                PlatformOpenGL._wglBindBuffer(WebGL2RenderingContext.UNIFORM_BUFFER, null);
   488→                return buffer;
   489→        }
   490→
   491→        /**
   492→         * Updates the data in a uniform buffer object.

   501→                PlatformOpenGL._wglBindBuffer(WebGL2RenderingContext.UNIFORM_BUFFER, null);

   523→        // ========== Shader Validation ==========
   524→
   525→        /**
   526→         * Validates a linked shader program.
   527→         * Useful for debugging; not recommended for production use due to performance cost.
   528→         *
   529→         * @param program the program to validate
   530→         * @return true if the program validated successfully
   531→         */
   532→        public static boolean validateProgram(JSObject program) {
   533→                if (program == null) return false;
   534→                PlatformOpenGL._wglValidateProgram(program);
   535→                // Validation result is informational; the program may still work even if
   536→                // validation "fails" (e.g., current vertex state doesn't match program)
   537→                return true;
   538→        }

   611→        // ========== Title Screen Renderer ==========
   612→
   613→        /** Cached title screen shader program. */
   614→        private static JSObject titleProgram = null;
   615→
   616→        /** Cached title screen VAO. */
   617→        private static JSObject titleVAO = null;
   618→
   619→        /** Cached title screen VBO. */
   620→        private static JSObject titleVBO = null;
   621→
   622→        /** Cached uniform locations for the title shader. */
   623→        private static JSObject titleResolutionUniform = null;
   624→        private static JSObject titleTimeUniform = null;
   625→
   626→        /** Whether the title screen resources have been initialized. */
   627→        private static boolean titleInitialized = false;
   628→
   629→        /**
   630→         * Renders the EaglerCraft title screen using a fullscreen gradient shader.
   631→         *
   632→         * <p>This draws a Minecraft-style sky gradient with a slowly moving
   633→         * panorama effect, plus overlaid text info. It serves as the placeholder
   634→         * title screen until the full MC GUI system is integrated.</p>
   635→         *
   636→         * @param gl       the WebGL2 context
   637→         * @param width    the canvas width in pixels
   638→         * @param height   the canvas height in pixels
   639→         * @param title    the main title text (e.g. "EaglerCraftX 26.1.2")
   640→         * @param subtitle the subtitle text
   641→         * @param username the player username
   642→         * @param seconds  elapsed seconds
   643→         * @param frames   total frames rendered
   644→         */
   645→        public static void renderTitleScreen(WebGL2RenderingContext gl, int width, int height,
   646→                        String title, String subtitle, String username, int seconds, int frames) {
   647→                if (!titleInitialized) {
   648→                        initTitleResources(gl);
   649→                        titleInitialized = true;
   650→                }
   651→                if (titleProgram == null) return;
   652→
   653→                // Use the title screen shader
   654→                PlatformOpenGL._wglUseProgram(titleProgram);
   655→
   656→                // Set uniforms
   657→                PlatformOpenGL._wglUniform2f(titleResolutionUniform, (float)width, (float)height);
   658→                PlatformOpenGL._wglUniform1f(titleTimeUniform, (float)seconds);
   659→
   660→                // Draw fullscreen quad
   661→                PlatformOpenGL._wglBindVertexArray(titleVAO);
   662→                gl.drawArrays(WebGL2RenderingContext.TRIANGLE_STRIP, 0, 4);
   663→                PlatformOpenGL._wglBindVertexArray(null);
   664→
   665→                // Unbind shader
   666→                PlatformOpenGL._wglUseProgram(null);
   667→        }
   668→
   669→        /**
   670→         * Initializes the WebGL2 resources needed for the title screen renderer.
   671→         * Creates a gradient shader program and a fullscreen quad VAO.
   672→         */
   673→        private static void initTitleResources(WebGL2RenderingContext gl) {
   674→                // Vertex shader: fullscreen triangle strip quad
   675→                String vsh = "#version 300 es\n"
   676→                        + "precision highp float;\n"
   677→                        + "in vec2 a_pos;\n"
   678→                        + "out vec2 v_uv;\n"
   679→                        + "void main() {\n"
   680→                        + "  v_uv = a_pos * 0.5 + 0.5;\n"
   681→                        + "  gl_Position = vec4(a_pos, 0.0, 1.0);\n"
   682→                        + "}\n";
   683→
   684→                // Fragment shader: MC-style sky gradient with animated panorama
   685→                String fsh = "#version 300 es\n"
   686→                        + "precision highp float;\n"
   687→                        + "in vec2 v_uv;\n"
   688→                        + "uniform vec2 u_resolution;\n"
   689→                        + "uniform float u_time;\n"
   690→                        + "out vec4 fragColor;\n"
   691→                        + "void main() {\n"
   692→                        + "  vec2 uv = v_uv;\n"
   693→                        + "  float horizon = 0.35;\n"
   694→                        + "  // Sky gradient (MC overworld sky colors)\n"
   695→                        + "  vec3 skyTop = vec3(0.38, 0.62, 0.95);\n"
   696→                        + "  vec3 skyHorizon = vec3(0.75, 0.87, 1.0);\n"
   697→                        + "  vec3 groundFar = vec3(0.55, 0.73, 0.35);\n"
   698→                        + "  vec3 groundNear = vec3(0.38, 0.55, 0.22);\n"
   699→                        + "  vec3 col;\n"
   700→                        + "  if (uv.y > horizon) {\n"
   701→                        + "    float t = (uv.y - horizon) / (1.0 - horizon);\n"
   702→                        + "    col = mix(skyHorizon, skyTop, t);\n"
   703→                        + "    // Subtle clouds\n"
   704→                        + "    float cloud = smoothstep(0.48, 0.52, sin(uv.x * 8.0 + u_time * 0.15) * 0.5 + 0.5);\n"
   705→                        + "    col = mix(col, vec3(1.0), cloud * 0.3 * smoothstep(0.0, 0.3, t));\n"
   706→                        + "  } else {\n"
   707→                        + "    float t = uv.y / horizon;\n"
   708→                        + "    col = mix(groundNear, groundFar, t);\n"
   709→                        + "  }\n"
   710→                        + "  // Vignette effect\n"
   711→                        + "  vec2 vc = uv - 0.5;\n"
   712→                        + "  float vig = 1.0 - dot(vc, vc) * 0.8;\n"
   713→                        + "  col *= vig;\n"
   714→                        + "  fragColor = vec4(col, 1.0);\n"
   715→                        + "}\n";
   716→
   717→                titleProgram = createProgramFromSource(vsh, fsh, "title_screen");
   718→                if (titleProgram == null) {
   719→                        ClientMain.error("[EaglerShaderImpl] Failed to create title screen shader");
   720→                        return;
   721→                }
   722→
   723→                // Get uniform locations
   724→                titleResolutionUniform = getUniformLocation(titleProgram, "u_resolution");
   725→                titleTimeUniform = getUniformLocation(titleProgram, "u_time");
   726→
   727→                // Create fullscreen quad VAO
   728→                titleVAO = PlatformOpenGL._wglCreateVertexArray();
   729→                PlatformOpenGL._wglBindVertexArray(titleVAO);
   730→
   731→                titleVBO = PlatformOpenGL._wglCreateBuffer();
   732→                PlatformOpenGL._wglBindBuffer(WebGL2RenderingContext.ARRAY_BUFFER, titleVBO);
   733→
   734→                // Fullscreen quad vertices (triangle strip): -1,-1  1,-1  -1,1  1,1
   735→                float[] quadData = new float[] {
   736→                        -1.0f, -1.0f,
   737→                         1.0f, -1.0f,
   738→                        -1.0f,  1.0f,
   739→                         1.0f,  1.0f
   740→                };
   741→                Float32Array quadBuf = Float32Array.create(quadData.length);
   742→                for (int i = 0; i < quadData.length; i++) {
   743→                        quadBuf.set(i, quadData[i]);
   744→                }
   745→                PlatformOpenGL._wglBufferData(WebGL2RenderingContext.ARRAY_BUFFER, quadBuf,
   746→                        WebGL2RenderingContext.STATIC_DRAW);
   747→
   748→                int posAttrib = getAttribLocation(titleProgram, "a_pos");
   749→                PlatformOpenGL._wglEnableVertexAttribArray(posAttrib);
   750→                PlatformOpenGL._wglVertexAttribPointer(posAttrib, 2,
   751→                        WebGL2RenderingContext.FLOAT, false, 0, 0);
   752→
   753→                PlatformOpenGL._wglBindVertexArray(null);
   754→                PlatformOpenGL._wglBindBuffer(WebGL2RenderingContext.ARRAY_BUFFER, null);
   755→
   756→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   757→                        ClientMain.log("[EaglerShaderImpl] Title screen shader initialized");
   758→                }
   759→        }
   760→}
