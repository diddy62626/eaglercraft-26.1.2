    14→/**
    15→ * Main game entry point adapter for EaglerCraft 26.1.2.
    16→ *
    17→ * <p>This class bridges the TeaVM platform layer (initialized by
    18→ * {@link ClientMain}) with Minecraft 26.1.2's game code. It is called
    19→ * after the platform subsystems are ready and is responsible for:</p>
    20→ * <ol>
    21→ *   <li>Initializing the EaglerCraft adapter layer (config, profile, services)</li>
    22→ *   <li>Setting up the WebGL2 rendering pipeline for MC's renderer</li>
    23→ *   <li>Creating the MC Minecraft instance and starting the game loop</li>
    24→ *   <li>Handling frame-by-frame tick/render dispatch</li>
    25→ *   <li>Signaling readiness via {@link ClientMain#__eaglercraftReady()}</li>
    26→ * </ol>
    27→ *
    28→ * <h3>Initialization sequence:</h3>
    29→ * <pre>
    30→ *   1. MainClass.main()           → TeaVM entry point
    31→ *   2. ClientMain._main()         → Platform init (canvas, WebGL2, input, audio, etc.)
    32→ *   3. ClientMain game loop       → requestAnimationFrame tick
    33→ *   4. EaglerCraft.initialize()   ← Called from the game loop after platform is ready
    34→ *      a. EaglerCraftConfig       → Constants loaded (no-op, all static final)
    35→ *      b. EaglerProfile.load()    → Load saved settings from localStorage
    36→ *      c. PlatformWebService.init() → Verify service availability
    37→ *      d. EaglerShaderImpl        → Reset shader statistics
    38→ *      e. Minecraft instance      → Create the MC game instance
    39→ *      f. ClientMain.__eaglercraftReady() → Signal browser that game is ready
    40→ *   5. EaglerCraft.tick()         ← Called each frame from the game loop
    41→ *      a. Process input events
    42→ *      b. Run MC game tick (logic update at 20 TPS)
    43→ *      c. Render frame
    44→ * </pre>
    45→ *
    46→ * <h3>Architecture notes:</h3>
    47→ * <p>This class is in the <code>main</code> source set, not the <code>teavm</code>
    48→ * source set. The <code>teavm</code> source set handles low-level browser platform
    49→ * concerns (WebGL2 context, DOM events, IndexedDB, etc.), while this class
    50→ * operates at the MC game level, bridging between MC's architecture and the
    51→ * browser platform.</p>
    52→ *

    54→ * should be updated to actual MC class names. Expected patterns:</p>
    55→ * <ul>

   202→        /**
   203→         * Sets up the WebGL2 rendering pipeline for Minecraft's renderer.
   204→         *
   205→         * <p>Configures the initial GL state that MC's GameRenderer expects:</p>
   206→         * <ul>
   207→         *   <li>Set viewport to canvas size</li>
   208→         *   <li>Enable depth testing</li>
   209→         *   <li>Enable back-face culling</li>
   210→         *   <li>Set up the clear color</li>
   211→         *   <li>Configure blending for transparency</li>
   212→         * </ul>
   213→         */
   214→        private static void setupRenderingPipeline() {
   215→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   216→                        ClientMain.log("[EaglerCraft] Setting up rendering pipeline...");
   217→                }
   218→
   219→                // Set initial viewport
   220→                int width = PlatformRuntime.getCanvasDrawableWidth();
   221→                int height = PlatformRuntime.getCanvasDrawableHeight();
   222→                PlatformOpenGL._wglViewport(0, 0, width, height);
   223→
   224→                // Enable depth testing (MC requires this)
   225→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.DEPTH_TEST);
   226→                PlatformOpenGL._wglDepthFunc(WebGL2RenderingContext.LEQUAL);
   227→
   228→                // Enable back-face culling (MC uses this for block faces)
   229→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.CULL_FACE);
   230→                PlatformOpenGL._wglCullFace(WebGL2RenderingContext.BACK);
   231→                PlatformOpenGL._wglFrontFace(WebGL2RenderingContext.CCW);
   232→
   233→                // Set default clear color (sky blue-ish, matches MC default)
   234→                PlatformOpenGL._wglClearColor(0.627f, 0.808f, 1.0f, 1.0f);
   235→
   236→                // Configure blending for transparency (used heavily by MC)
   237→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.BLEND);
   238→                PlatformOpenGL._wglBlendFunc(WebGL2RenderingContext.SRC_ALPHA,
   239→                                WebGL2RenderingContext.ONE_MINUS_SRC_ALPHA);
   240→
   241→                // Disable stencil test by default (MC uses it for some effects)
   242→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.STENCIL_TEST);
   243→
   244→                // Enable scissor test (used by MC for viewport clipping)
   245→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.SCISSOR_TEST);
   246→
   247→                // Reset shader stats
   248→                EaglerShaderImpl.resetStats();
   249→
   250→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   251→                        ClientMain.log("[EaglerCraft] Rendering pipeline configured "
   252→                                        + "(viewport: " + width + "x" + height + ")");
   253→                        ClientMain.log("[EaglerCraft] WebGL2 Version: " + PlatformOpenGL.getGLVersion());
   254→                        ClientMain.log("[EaglerCraft] GL Vendor: " + PlatformOpenGL.getGLVendor());
   255→                        ClientMain.log("[EaglerCraft] GL Renderer: " + PlatformOpenGL.getGLRenderer());
   256→                }
   257→        }

   360→        /**
   361→         * Called once per frame by the ClientMain game loop.
   362→         *
   363→         * <p>This method handles the frame-by-frame update cycle:</p>
   364→         * <ol>
   365→         *   <li>Check if the page is visible (auto-pause when hidden)</li>
   366→         *   <li>Process input events from PlatformInput</li>
   367→         *   <li>Run game logic ticks at 20 TPS (fixed timestep)</li>
   368→         *   <li>Render the frame</li>
   369→         *   <li>Update FPS counter and timing</li>
   370→         * </ol>
   371→         *
   372→         * <p>The tick rate is decoupled from the frame rate: the game logic
   373→         * runs at a fixed 20 TPS, while rendering happens at the display's
   374→         * refresh rate (typically 60Hz via requestAnimationFrame). This
   375→         * prevents simulation instability from variable frame timing.</p>
   376→         *
   377→         * <p>TODO: Once MC 26.1.2 is decompiled, this method should be
   378→         * patched to call the actual MC tick/render methods. Expected:</p>
   379→         * <pre>
   380→         *   Minecraft mc = Minecraft.getInstance();
   381→         *   mc.runTick(isPaused);
   382→         * </pre>
   383→         */
   384→        public static void tick() {
   385→                if (gameState != STATE_RUNNING) return;
   386→
   387→                long currentTime = PlatformRuntime.getCurrentTimeMillis();
   388→                long deltaTime = currentTime - lastTickTime;
   389→
   390→                // Clamp delta time to prevent spiral of death
   391→                if (deltaTime > EaglerCraftConfig.MAX_FRAME_DELTA_MS) {
   392→                        deltaTime = EaglerCraftConfig.MAX_FRAME_DELTA_MS;
   393→                }
   394→
   395→                // Auto-pause when page is not visible
   396→                if (!PlatformRuntime.isPageVisible()) {
   397→                        if (!paused) {
   398→                                paused = true;
   399→                                gameState = STATE_PAUSED;
   400→                        }
   401→                        return;
   402→                } else if (paused) {
   403→                        paused = false;
   404→                        gameState = STATE_RUNNING;
   405→                        lastTickTime = currentTime;
   406→                        tickAccumulator = 0;
   407→                        return;
   408→                }
   409→
   410→                // Process input
   411→                processInput();
   412→
   413→                // Run game logic ticks at fixed TPS
   414→                if (EaglerCraftConfig.USE_FIXED_TIMESTEP) {
   415→                        tickAccumulator += deltaTime;
   416→                        while (tickAccumulator >= EaglerCraftConfig.MS_PER_TICK) {
   417→                                gameLogicTick();
   418→                                tickAccumulator -= EaglerCraftConfig.MS_PER_TICK;
   419→                        }
   420→                } else {
   421→                        // Variable timestep: run one tick per frame
   422→                        // MC internally handles the tick rate, so we just call it
   423→                        gameLogicTick();
   424→                }
   425→
   426→                // Render frame
   427→                renderFrame();
   428→
   429→                // Update timing
   430→                lastTickTime = currentTime;
   431→                totalFrames++;
   432→        }
