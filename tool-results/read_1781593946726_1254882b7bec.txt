   203→   203→         * Sets up the WebGL2 rendering pipeline for Minecraft's renderer.
   204→   204→         *
   205→   205→         * <p>Configures the initial GL state that MC's GameRenderer expects:</p>
   206→   206→         * <ul>
   207→   207→         *   <li>Set viewport to canvas size</li>
   208→   208→         *   <li>Enable depth testing</li>
   209→   209→         *   <li>Enable back-face culling</li>
   210→   210→         *   <li>Set up the clear color</li>
   211→   211→         *   <li>Configure blending for transparency</li>
   212→   212→         * </ul>
   213→   213→         */
   214→   214→        private static void setupRenderingPipeline() {
   215→   215→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   216→   216→                        ClientMain.log("[EaglerCraft] Setting up rendering pipeline...");
   217→   217→                }
   218→   218→
   219→   219→                // Set initial viewport
   220→   220→                int width = PlatformRuntime.getCanvasDrawableWidth();
   221→   221→                int height = PlatformRuntime.getCanvasDrawableHeight();
   222→   222→                PlatformOpenGL._wglViewport(0, 0, width, height);
   223→   223→
   224→   224→                // Enable depth testing (MC requires this)
   225→   225→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.DEPTH_TEST);
   226→   226→                PlatformOpenGL._wglDepthFunc(WebGL2RenderingContext.LEQUAL);
   227→   227→
   228→   228→                // Enable back-face culling (MC uses this for block faces)
   229→   229→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.CULL_FACE);
   230→   230→                PlatformOpenGL._wglCullFace(WebGL2RenderingContext.BACK);
   231→   231→                PlatformOpenGL._wglFrontFace(WebGL2RenderingContext.CCW);
   232→   232→
   233→   233→                // Set default clear color (sky blue-ish, matches MC default)
   234→   234→                PlatformOpenGL._wglClearColor(0.627f, 0.808f, 1.0f, 1.0f);
   235→   235→
   236→   236→                // Configure blending for transparency (used heavily by MC)
   237→   237→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.BLEND);
   238→   238→                PlatformOpenGL._wglBlendFunc(WebGL2RenderingContext.SRC_ALPHA,
   239→   239→                                WebGL2RenderingContext.ONE_MINUS_SRC_ALPHA);
   240→   240→
   241→   241→                // Disable stencil test by default (MC uses it for some effects)
   242→   242→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.STENCIL_TEST);
   243→   243→
   244→   244→                // Enable scissor test (used by MC for viewport clipping)
   245→   245→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.SCISSOR_TEST);
   246→   246→
   247→   247→                // Reset shader stats
   248→   248→                EaglerShaderImpl.resetStats();
   249→   249→
   250→   250→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   251→   251→                        ClientMain.log("[EaglerCraft] Rendering pipeline configured "
   252→   252→                                        + "(viewport: " + width + "x" + height + ")");
   253→   253→                        ClientMain.log("[EaglerCraft] WebGL2 Version: " + PlatformOpenGL.getGLVersion());
   254→   254→                        ClientMain.log("[EaglerCraft] GL Vendor: " + PlatformOpenGL.getGLVendor());
   255→   255→                        ClientMain.log("[EaglerCraft] GL Renderer: " + PlatformOpenGL.getGLRenderer());
   256→   256→                }
   257→   257→        }

   285→   285→         *   <li>Asset root from the EPK loader</li>
   286→   286→         * </ul>
   287→   287→         *
   288→   288→         * <p>TODO: Update to actual MC 26.1.2 API after decompilation.
   289→   289→         * The exact constructor signature and initialization sequence will
   290→   290→         * depend on the MC 26.1.2 codebase. Expected pattern:</p>

   358→   358→        // ========== Game Loop ==========
   359→   359→
   360→   360→        /**
   361→   361→         * Called once per frame by the ClientMain game loop.
   362→   362→         *
   363→   363→         * <p>This method handles the frame-by-frame update cycle:</p>
   364→   364→         * <ol>
   365→   365→         *   <li>Check if the page is visible (auto-pause when hidden)</li>
   366→   366→         *   <li>Process input events from PlatformInput</li>
   367→   367→         *   <li>Run game logic ticks at 20 TPS (fixed timestep)</li>
   368→   368→         *   <li>Render the frame</li>
   369→   369→         *   <li>Update FPS counter and timing</li>
   370→   370→         * </ol>
   371→   371→         *
   372→   372→         * <p>The tick rate is decoupled from the frame rate: the game logic
   373→   373→         * runs at a fixed 20 TPS, while rendering happens at the display's
   374→   374→         * refresh rate (typically 60Hz via requestAnimationFrame). This
   375→   375→         * prevents simulation instability from variable frame timing.</p>
   376→   376→         *
   377→   377→         * <p>TODO: Once MC 26.1.2 is decompiled, this method should be
   378→   378→         * patched to call the actual MC tick/render methods. Expected:</p>
   379→   379→         * <pre>
   380→   380→         *   Minecraft mc = Minecraft.getInstance();
   381→   381→         *   mc.runTick(isPaused);
   382→   382→         * </pre>
   383→   383→         */
   384→   384→        public static void tick() {
   385→   385→                if (gameState != STATE_RUNNING) return;
   386→   386→
   387→   387→                long currentTime = PlatformRuntime.getCurrentTimeMillis();
   388→   388→                long deltaTime = currentTime - lastTickTime;
   389→   389→
   390→   390→                // Clamp delta time to prevent spiral of death
   391→   391→                if (deltaTime > EaglerCraftConfig.MAX_FRAME_DELTA_MS) {
   392→   392→                        deltaTime = EaglerCraftConfig.MAX_FRAME_DELTA_MS;
   393→   393→                }
   394→   394→
   395→   395→                // Auto-pause when page is not visible
   396→   396→                if (!PlatformRuntime.isPageVisible()) {
   397→   397→                        if (!paused) {
   398→   398→                                paused = true;
   399→   399→                                gameState = STATE_PAUSED;
   400→   400→                        }
   401→   401→                        return;
   402→   402→                } else if (paused) {
   403→   403→                        paused = false;
   404→   404→                        gameState = STATE_RUNNING;
   405→   405→                        lastTickTime = currentTime;
   406→   406→                        tickAccumulator = 0;
   407→   407→                        return;
   408→   408→                }
   409→   409→
   410→   410→                // Process input
   411→   411→                processInput();
   412→   412→
   413→   413→                // Run game logic ticks at fixed TPS
   414→   414→                if (EaglerCraftConfig.USE_FIXED_TIMESTEP) {
   415→   415→                        tickAccumulator += deltaTime;
   416→   416→                        while (tickAccumulator >= EaglerCraftConfig.MS_PER_TICK) {
   417→   417→                                gameLogicTick();
   418→   418→                                tickAccumulator -= EaglerCraftConfig.MS_PER_TICK;
   419→   419→                        }
   420→   420→                } else {
   421→   421→                        // Variable timestep: run one tick per frame
   422→   422→                        // MC internally handles the tick rate, so we just call it
   423→   423→                        gameLogicTick();
   424→   424→                }
   425→   425→
   426→   426→                // Render frame
   427→   427→                renderFrame();
