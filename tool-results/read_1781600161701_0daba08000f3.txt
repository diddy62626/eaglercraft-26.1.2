    14→/**
    15→ * Main game entry point adapter for EaglerCraft 26.1.2.
    16→ *
    17→ * <p>This class bridges the TeaVM platform layer (initialized by
    18→ * {@link ClientMain}) with Minecraft 26.1.2's game code. It is called
    19→ * after the platform subsystems are ready and is responsible for:</p>
    20→ * <ol>
    21→ *   <li>Initializing the EaglerCraft adapter layer (config, profile, services)</li>
    22→ *   <li>Setting up the WebGL2 rendering pipeline for MC's renderer</li>
    23→ *   <li>Creating the MC Minecraft instance and starting the game loop</li>
    24→ *   <li>Handling frame-by-frame tick/render dispatch</li>
    25→ *   <li>Signaling readiness via {@link ClientMain#__eaglercraftReady()}</li>
    26→ * </ol>
    27→ *
    28→ * <h3>Initialization sequence:</h3>
    29→ * <pre>
    30→ *   1. MainClass.main()           → TeaVM entry point
    31→ *   2. ClientMain._main()         → Platform init (canvas, WebGL2, input, audio, etc.)
    32→ *   3. ClientMain game loop       → requestAnimationFrame tick
    33→ *   4. EaglerCraft.initialize()   ← Called from the game loop after platform is ready
    34→ *      a. EaglerCraftConfig       → Constants loaded (no-op, all static final)
    35→ *      b. EaglerProfile.load()    → Load saved settings from localStorage
    36→ *      c. PlatformWebService.init() → Verify service availability
    37→ *      d. EaglerShaderImpl        → Reset shader statistics
    38→ *      e. Minecraft instance      → Create the MC game instance
    39→ *      f. ClientMain.__eaglercraftReady() → Signal browser that game is ready
    40→ *   5. EaglerCraft.tick()         ← Called each frame from the game loop
    41→ *      a. Process input events
    42→ *      b. Run MC game tick (logic update at 20 TPS)
    43→ *      c. Render frame
    44→ * </pre>
    45→ *
    46→ * <h3>Architecture notes:</h3>
    47→ * <p>This class is in the <code>main</code> source set, not the <code>teavm</code>
    48→ * source set. The <code>teavm</code> source set handles low-level browser platform
    49→ * concerns (WebGL2 context, DOM events, IndexedDB, etc.), while this class
    50→ * operates at the MC game level, bridging between MC's architecture and the
    51→ * browser platform.</p>
    52→ *

    54→ * should be updated to actual MC class names. Expected patterns:</p>

    68→         * Generates an offline-mode UUID from a username.
    69→         * Replaces UUID.nameUUIDFromBytes which TeaVM doesn't support.
    70→         * Uses the same algorithm as Mojang's offline UUID:
    71→         * UUID v3 with namespace "OfflinePlayer:" prefix.
    72→         */
    73→        private static java.util.UUID makeOfflineUUID(String username) {
    74→                // TeaVM doesn't support UUID(long, long) constructor.
    75→                // Use UUID.fromString() with a deterministic format instead.
    76→                // This is a simplified offline UUID - not cryptographically compatible
    77→                // with Mojang's offline UUID but sufficient for EaglerCraft.
    78→                int hash = username.hashCode();
    79→                return java.util.UUID.fromString(
    80→                        String.format("%08x-0000-3000-8000-%012x",
    81→                                hash, (long)hash * 31 & 0xFFFFFFFFFFFFL)
    82→                );
    83→        }

   220→        /**
   221→         * Sets up the WebGL2 rendering pipeline for Minecraft's renderer.
   222→         *
   223→         * <p>Configures the initial GL state that MC's GameRenderer expects:</p>
   224→         * <ul>
   225→         *   <li>Set viewport to canvas size</li>
   226→         *   <li>Enable depth testing</li>
   227→         *   <li>Enable back-face culling</li>
   228→         *   <li>Set up the clear color</li>
   229→         *   <li>Configure blending for transparency</li>
   230→         * </ul>
   231→         */
   232→        private static void setupRenderingPipeline() {
   233→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   234→                        ClientMain.log("[EaglerCraft] Setting up rendering pipeline...");
   235→                }
   236→
   237→                // Set initial viewport
   238→                int width = PlatformRuntime.getCanvasDrawableWidth();
   239→                int height = PlatformRuntime.getCanvasDrawableHeight();
   240→                PlatformOpenGL._wglViewport(0, 0, width, height);
   241→
   242→                // Enable depth testing (MC requires this)
   243→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.DEPTH_TEST);
   244→                PlatformOpenGL._wglDepthFunc(WebGL2RenderingContext.LEQUAL);
   245→
   246→                // Enable back-face culling (MC uses this for block faces)
   247→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.CULL_FACE);
   248→                PlatformOpenGL._wglCullFace(WebGL2RenderingContext.BACK);
   249→                PlatformOpenGL._wglFrontFace(WebGL2RenderingContext.CCW);
   250→
   251→                // Set default clear color (sky blue-ish, matches MC default)
   252→                PlatformOpenGL._wglClearColor(0.627f, 0.808f, 1.0f, 1.0f);
   253→
   254→                // Configure blending for transparency (used heavily by MC)
   255→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.BLEND);
   256→                PlatformOpenGL._wglBlendFunc(WebGL2RenderingContext.SRC_ALPHA,
   257→                                WebGL2RenderingContext.ONE_MINUS_SRC_ALPHA);
   258→
   259→                // Disable stencil test by default (MC uses it for some effects)
   260→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.STENCIL_TEST);
   261→
   262→                // Enable scissor test (used by MC for viewport clipping)
   263→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.SCISSOR_TEST);
   264→
   265→                // Reset shader stats
   266→                EaglerShaderImpl.resetStats();
   267→
   268→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   269→                        ClientMain.log("[EaglerCraft] Rendering pipeline configured "
   270→                                        + "(viewport: " + width + "x" + height + ")");
   271→                        ClientMain.log("[EaglerCraft] WebGL2 Version: " + PlatformOpenGL.getGLVersion());
   272→                        ClientMain.log("[EaglerCraft] GL Vendor: " + PlatformOpenGL.getGLVendor());
   273→                        ClientMain.log("[EaglerCraft] GL Renderer: " + PlatformOpenGL.getGLRenderer());
   274→                }
   275→        }

   396→        /**
   397→         * Called once per frame by the ClientMain game loop.
   398→         *
   399→         * <p>This method handles the frame-by-frame update cycle:</p>
   400→         * <ol>
   401→         *   <li>Check if the page is visible (auto-pause when hidden)</li>
   402→         *   <li>Process input events from PlatformInput</li>
   403→         *   <li>Run game logic ticks at 20 TPS (fixed timestep)</li>
   404→         *   <li>Render the frame</li>
   405→         *   <li>Update FPS counter and timing</li>
   406→         * </ol>
   407→         *
   408→         * <p>The tick rate is decoupled from the frame rate: the game logic
   409→         * runs at a fixed 20 TPS, while rendering happens at the display's
   410→         * refresh rate (typically 60Hz via requestAnimationFrame). This
   411→         * prevents simulation instability from variable frame timing.</p>
   412→         *
   413→         * <p>TODO: Once MC 26.1.2 is decompiled, this method should be
   414→         * patched to call the actual MC tick/render methods. Expected:</p>
   415→         * <pre>
   416→         *   Minecraft mc = Minecraft.getInstance();
   417→         *   mc.runTick(isPaused);
   418→         * </pre>
   419→         */
   420→        public static void tick() {
   421→                if (gameState != STATE_RUNNING) return;
   422→
   423→                long currentTime = PlatformRuntime.getCurrentTimeMillis();
   424→                long deltaTime = currentTime - lastTickTime;
   425→
   426→                // Clamp delta time to prevent spiral of death
   427→                if (deltaTime > EaglerCraftConfig.MAX_FRAME_DELTA_MS) {
   428→                        deltaTime = EaglerCraftConfig.MAX_FRAME_DELTA_MS;
   429→                }
   430→
   431→                // Auto-pause when page is not visible
   432→                if (!PlatformRuntime.isPageVisible()) {
   433→                        if (!paused) {
   434→                                paused = true;
   435→                                gameState = STATE_PAUSED;
   436→                        }
   437→                        return;
   438→                } else if (paused) {
   439→                        paused = false;
   440→                        gameState = STATE_RUNNING;
   441→                        lastTickTime = currentTime;
   442→                        tickAccumulator = 0;
   443→                        return;
   444→                }
   445→
   446→                // Process input
   447→                processInput();
   448→
   449→                // Run game logic ticks at fixed TPS
   450→                if (EaglerCraftConfig.USE_FIXED_TIMESTEP) {
   451→                        tickAccumulator += deltaTime;
   452→                        while (tickAccumulator >= EaglerCraftConfig.MS_PER_TICK) {
   453→                                gameLogicTick();
   454→                                tickAccumulator -= EaglerCraftConfig.MS_PER_TICK;
   455→                        }
   456→                } else {
   457→                        // Variable timestep: run one tick per frame
   458→                        // MC internally handles the tick rate, so we just call it
   459→                        gameLogicTick();
   460→                }
   461→
   462→                // Render frame
   463→                renderFrame();
   464→
   465→                // Update timing
   466→                lastTickTime = currentTime;
   467→                totalFrames++;
   468→        }

   514→                        try {
   515→                                net.minecraft.client.Minecraft mc = (net.minecraft.client.Minecraft) minecraftInstance;
   516→                                mc.tick();
   517→                        } catch (Throwable t) {
   518→                                // MC tick errors shouldn't crash the game loop
   519→                                ClientMain.warn("[EaglerCraft] MC tick error: " + t.getMessage());
   520→                        }
   521→                }
   522→                totalTicks++;
   523→        }
   524→
   525→        /**
   526→         * Renders one frame.
   527→         *
   528→         * <p>MC's rendering pipeline is driven by GameRenderer, which
   529→         * handles world rendering, GUI overlay, and screen rendering.</p>
   530→         *
   531→         * <p>Currently renders a Minecraft-style sky background with a
   532→         * title screen until the full MC renderer is integrated.</p>
   533→         */
   534→        private static void renderFrame() {
   535→                // Check for viewport changes
   536→                int width = PlatformRuntime.getCanvasDrawableWidth();
   537→                int height = PlatformRuntime.getCanvasDrawableHeight();
   538→                if (width <= 0 || height <= 0) return;
   539→
   540→                // If MC is running, its own game loop handles rendering via run()
   541→                // The EaglerCraft game loop is separate from MC's internal loop.
   542→                // MC's run() method has its own render loop.
   543→                // We only render our fallback title screen if MC isn't running.
   544→
   545→                if (minecraftInstance instanceof net.minecraft.client.Minecraft) {
   546→                        // MC handles its own rendering through its game loop.
   547→                        // The MC instance's run() method manages tick + render.
   548→                        // We just need to clear the framebuffer as a safety net.
   549→                        PlatformOpenGL._wglClear(
   550→                                WebGL2RenderingContext.COLOR_BUFFER_BIT
   551→                                | WebGL2RenderingContext.DEPTH_BUFFER_BIT);
   552→                        return;
   553→                }
   554→
   555→                // Fallback: render the title screen using WebGL2
   556→                PlatformOpenGL._wglClear(
   557→                        WebGL2RenderingContext.COLOR_BUFFER_BIT
   558→                        | WebGL2RenderingContext.DEPTH_BUFFER_BIT);
   559→
   560→                // Render the title screen using WebGL2
   561→                renderTitleScreen(width, height);
   562→
   563→                firstFrameRendered = true;
   564→        }
   565→
   566→        /**
   567→         * Renders the EaglerCraft title screen directly via WebGL2.
   568→         *
   569→         * <p>This is a placeholder title screen rendered before the full
   570→         * MC GUI system is integrated. It draws:</p>
   571→         * <ul>
   572→         *   <li>A gradient sky background (MC-style)</li>
   573→         *   <li>The EaglerCraftX title text</li>
   574→         *   <li>Version info and status</li>
   575→         * </ul>
   576→         */
   577→        private static void renderTitleScreen(int width, int height) {
   578→                WebGL2RenderingContext gl = ClientMain.getWebGL2();
   579→                if (gl == null) return;
   580→
   581→                // Use a simple fullscreen quad with a gradient shader
   582→                // to render the title screen background
   583→                EaglerShaderImpl.renderTitleScreen(gl, width, height,
   584→                        EaglerCraftConfig.CLIENT_IDENTIFIER,
   585→                        "Protocol " + EaglerCraftConfig.PROTOCOL_VERSION,
   586→                        EaglerProfile.getUsername(),
   587→                        (int)(totalTicks / 20),
   588→                        (int)totalFrames);
   589→        }
