    17→    17→ * <p>This class bridges the TeaVM platform layer (initialized by

    41→    41→ *      a. Process input events

    69→    69→         * Replaces UUID.nameUUIDFromBytes which TeaVM doesn't support.

    74→    74→                // TeaVM doesn't support UUID(long, long) constructor.
    75→    75→                // Use UUID.fromString() with a deterministic format instead.
    76→    76→                // This is a simplified offline UUID - not cryptographically compatible
    77→    77→                // with Mojang's offline UUID but sufficient for EaglerCraft.
    78→    78→                int hash = username.hashCode();
    79→    79→                return java.util.UUID.fromString(
    80→    80→                        String.format("%08x-0000-3000-8000-%012x",
    81→    81→                                hash, (long)hash * 31 & 0xFFFFFFFFFFFFL)
    82→    82→                );
    83→    83→        }

   229→   229→         *   <li>Configure blending for transparency</li>

   232→   232→        private static void setupRenderingPipeline() {
   233→   233→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   234→   234→                        ClientMain.log("[EaglerCraft] Setting up rendering pipeline...");
   235→   235→                }
   236→   236→
   237→   237→                // Set initial viewport
   238→   238→                int width = PlatformRuntime.getCanvasDrawableWidth();
   239→   239→                int height = PlatformRuntime.getCanvasDrawableHeight();
   240→   240→                PlatformOpenGL._wglViewport(0, 0, width, height);
   241→   241→
   242→   242→                // Enable depth testing (MC requires this)
   243→   243→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.DEPTH_TEST);
   244→   244→                PlatformOpenGL._wglDepthFunc(WebGL2RenderingContext.LEQUAL);
   245→   245→
   246→   246→                // Enable back-face culling (MC uses this for block faces)
   247→   247→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.CULL_FACE);
   248→   248→                PlatformOpenGL._wglCullFace(WebGL2RenderingContext.BACK);
   249→   249→                PlatformOpenGL._wglFrontFace(WebGL2RenderingContext.CCW);
   250→   250→
   251→   251→                // Set default clear color (sky blue-ish, matches MC default)
   252→   252→                PlatformOpenGL._wglClearColor(0.627f, 0.808f, 1.0f, 1.0f);
   253→   253→
   254→   254→                // Configure blending for transparency (used heavily by MC)
   255→   255→                PlatformOpenGL._wglEnable(WebGL2RenderingContext.BLEND);
   256→   256→                PlatformOpenGL._wglBlendFunc(WebGL2RenderingContext.SRC_ALPHA,
   257→   257→                                WebGL2RenderingContext.ONE_MINUS_SRC_ALPHA);
   258→   258→
   259→   259→                // Disable stencil test by default (MC uses it for some effects)
   260→   260→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.STENCIL_TEST);
   261→   261→
   262→   262→                // Enable scissor test (used by MC for viewport clipping)
   263→   263→                PlatformOpenGL._wglDisable(WebGL2RenderingContext.SCISSOR_TEST);
   264→   264→
   265→   265→                // Reset shader stats
   266→   266→                EaglerShaderImpl.resetStats();
   267→   267→
   268→   268→                if (EaglerCraftConfig.VERBOSE_INIT_LOGGING) {
   269→   269→                        ClientMain.log("[EaglerCraft] Rendering pipeline configured "
   270→   270→                                        + "(viewport: " + width + "x" + height + ")");
   271→   271→                        ClientMain.log("[EaglerCraft] WebGL2 Version: " + PlatformOpenGL.getGLVersion());
   272→   272→                        ClientMain.log("[EaglerCraft] GL Vendor: " + PlatformOpenGL.getGLVendor());
   273→   273→                        ClientMain.log("[EaglerCraft] GL Renderer: " + PlatformOpenGL.getGLRenderer());
   274→   274→                }
   275→   275→        }

   400→   400→         * <ol>
   401→   401→         *   <li>Check if the page is visible (auto-pause when hidden)</li>
   402→   402→         *   <li>Process input events from PlatformInput</li>
   403→   403→         *   <li>Run game logic ticks at 20 TPS (fixed timestep)</li>
   404→   404→         *   <li>Render the frame</li>
   405→   405→         *   <li>Update FPS counter and timing</li>
   406→   406→         * </ol>
   407→   407→         *
   408→   408→         * <p>The tick rate is decoupled from the frame rate: the game logic
   409→   409→         * runs at a fixed 20 TPS, while rendering happens at the display's
   410→   410→         * refresh rate (typically 60Hz via requestAnimationFrame). This
   411→   411→         * prevents simulation instability from variable frame timing.</p>
   412→   412→         *
   413→   413→         * <p>TODO: Once MC 26.1.2 is decompiled, this method should be
   414→   414→         * patched to call the actual MC tick/render methods. Expected:</p>

   417→   417→         *   mc.runTick(isPaused);
   418→   418→         * </pre>
   419→   419→         */
   420→   420→        public static void tick() {
   421→   421→                if (gameState != STATE_RUNNING) return;
   422→   422→
   423→   423→                long currentTime = PlatformRuntime.getCurrentTimeMillis();
   424→   424→                long deltaTime = currentTime - lastTickTime;
   425→   425→
   426→   426→                // Clamp delta time to prevent spiral of death
   427→   427→                if (deltaTime > EaglerCraftConfig.MAX_FRAME_DELTA_MS) {
   428→   428→                        deltaTime = EaglerCraftConfig.MAX_FRAME_DELTA_MS;
   429→   429→                }
   430→   430→
   431→   431→                // Auto-pause when page is not visible
   432→   432→                if (!PlatformRuntime.isPageVisible()) {
   433→   433→                        if (!paused) {
   434→   434→                                paused = true;
   435→   435→                                gameState = STATE_PAUSED;
   436→   436→                        }
   437→   437→                        return;
   438→   438→                } else if (paused) {
   439→   439→                        paused = false;
   440→   440→                        gameState = STATE_RUNNING;
   441→   441→                        lastTickTime = currentTime;
   442→   442→                        tickAccumulator = 0;
   443→   443→                        return;
   444→   444→                }
   445→   445→
   446→   446→                // Process input
   447→   447→                processInput();
   448→   448→
   449→   449→                // Run game logic ticks at fixed TPS
   450→   450→                if (EaglerCraftConfig.USE_FIXED_TIMESTEP) {
   451→   451→                        tickAccumulator += deltaTime;
   452→   452→                        while (tickAccumulator >= EaglerCraftConfig.MS_PER_TICK) {
   453→   453→                                gameLogicTick();
   454→   454→                                tickAccumulator -= EaglerCraftConfig.MS_PER_TICK;
   455→   455→                        }
   456→   456→                } else {
   457→   457→                        // Variable timestep: run one tick per frame
   458→   458→                        // MC internally handles the tick rate, so we just call it
   459→   459→                        gameLogicTick();
   460→   460→                }
   461→   461→
   462→   462→                // Render frame
   463→   463→                renderFrame();
   464→   464→
   465→   465→                // Update timing
   466→   466→                lastTickTime = currentTime;
   467→   467→                totalFrames++;
   468→   468→        }
   469→   469→
   470→   470→        /**
   471→   471→         * Processes input events from PlatformInput.
   472→   472→         *
   473→   473→         * <p>Polls the keyboard and mouse state from PlatformInput and
   474→   474→         * dispatches events to the MC input system.</p>
   475→   475→         *
   476→   476→         * <p>TODO: Once MC 26.1.2 is decompiled, this should feed events
   477→   477→         * into MC's KeyboardHandler and MouseHandler. Expected:</p>
   478→   478→         * <pre>
   479→   479→         *   Minecraft mc = Minecraft.getInstance();
   480→   480→         *   KeyboardHandler kb = mc.keyboardHandler;
   481→   481→         *   MouseHandler mh = mc.mouseHandler;
   482→   482→         *   // Feed polled events to handlers
   483→   483→         * </pre>
   484→   484→         */
   485→   485→        private static void processInput() {
   486→   486→                // Input polling is handled by PlatformInput, which updates
   487→   487→                // its internal state arrays on each DOM event. MC code reads
   488→   488→                // the state through PlatformWebService convenience methods.
   489→   489→                //
   490→   490→                // In the full implementation, we would also:
   491→   491→                // - Check for new key press events and queue them for MC's KeyboardListener
   492→   492→                // - Check for mouse button click events and queue them for MC's MouseListener
   493→   493→                // - Handle pointer lock state changes

   496→   496→        }

   498→   498→        /**
   499→   499→         * Runs one game logic tick.
   500→   500→         *
   501→   501→         * <p>MC's game logic runs at 20 TPS (ticks per second). Each tick
   502→   502→         * updates entity positions, processes block changes, runs redstone,
   503→   503→         * handles network packets, etc.</p>
   504→   504→         *
   505→   505→         * <p>TODO: Once MC 26.1.2 is decompiled, this should call:</p>
   506→   506→         * <pre>
   507→   507→         *   Minecraft mc = Minecraft.getInstance();
   508→   508→         *   mc.tick();
   509→   509→         *   // or: mc.runTick(false);
   510→   510→         * </pre>
   511→   511→         */
   512→   512→        private static void gameLogicTick() {
   513→   513→                if (minecraftInstance instanceof net.minecraft.client.Minecraft) {
   514→   514→                        try {
   515→   515→                                net.minecraft.client.Minecraft mc = (net.minecraft.client.Minecraft) minecraftInstance;
   516→   516→                                mc.tick();
   517→   517→                        } catch (Throwable t) {
   518→   518→                                // MC tick errors shouldn't crash the game loop
   519→   519→                                ClientMain.warn("[EaglerCraft] MC tick error: " + t.getMessage());
   520→   520→                        }
   521→   521→                }
   522→   522→                totalTicks++;
   523→   523→        }
   524→   524→
   525→   525→        /**
   526→   526→         * Renders one frame.
   527→   527→         *
   528→   528→         * <p>MC's rendering pipeline is driven by GameRenderer, which
   529→   529→         * handles world rendering, GUI overlay, and screen rendering.</p>
   530→   530→         *
   531→   531→         * <p>Currently renders a Minecraft-style sky background with a
   532→   532→         * title screen until the full MC renderer is integrated.</p>
   533→   533→         */
   534→   534→        private static void renderFrame() {
   535→   535→                // Check for viewport changes
   536→   536→                int width = PlatformRuntime.getCanvasDrawableWidth();
   537→   537→                int height = PlatformRuntime.getCanvasDrawableHeight();
   538→   538→                if (width <= 0 || height <= 0) return;
   539→   539→
   540→   540→                // If MC is running, its own game loop handles rendering via run()
   541→   541→                // The EaglerCraft game loop is separate from MC's internal loop.
   542→   542→                // MC's run() method has its own render loop.
   543→   543→                // We only render our fallback title screen if MC isn't running.
   544→   544→
   545→   545→                if (minecraftInstance instanceof net.minecraft.client.Minecraft) {
   546→   546→                        // MC handles its own rendering through its game loop.
   547→   547→                        // The MC instance's run() method manages tick + render.
   548→   548→                        // We just need to clear the framebuffer as a safety net.
   549→   549→                        PlatformOpenGL._wglClear(
   550→   550→                                WebGL2RenderingContext.COLOR_BUFFER_BIT
   551→   551→                                | WebGL2RenderingContext.DEPTH_BUFFER_BIT);
   552→   552→                        return;
   553→   553→                }
   554→   554→
   555→   555→                // Fallback: render the title screen using WebGL2
   556→   556→                PlatformOpenGL._wglClear(
   557→   557→                        WebGL2RenderingContext.COLOR_BUFFER_BIT
   558→   558→                        | WebGL2RenderingContext.DEPTH_BUFFER_BIT);
   559→   559→
   560→   560→                // Render the title screen using WebGL2
   561→   561→                renderTitleScreen(width, height);
   562→   562→
   563→   563→                firstFrameRendered = true;
   564→   564→        }
   565→   565→
   566→   566→        /**
   567→   567→         * Renders the EaglerCraft title screen directly via WebGL2.
   568→   568→         *
   569→   569→         * <p>This is a placeholder title screen rendered before the full
   570→   570→         * MC GUI system is integrated. It draws:</p>
   571→   571→         * <ul>
   572→   572→         *   <li>A gradient sky background (MC-style)</li>
   573→   573→         *   <li>The EaglerCraftX title text</li>
   574→   574→         *   <li>Version info and status</li>
   575→   575→         * </ul>
   576→   576→         */
   577→   577→        private static void renderTitleScreen(int width, int height) {
   578→   578→                WebGL2RenderingContext gl = ClientMain.getWebGL2();
   579→   579→                if (gl == null) return;
   580→   580→
   581→   581→                // Use a simple fullscreen quad with a gradient shader
   582→   582→                // to render the title screen background
   583→   583→                EaglerShaderImpl.renderTitleScreen(gl, width, height,
   584→   584→                        EaglerCraftConfig.CLIENT_IDENTIFIER,
   585→   585→                        "Protocol " + EaglerCraftConfig.PROTOCOL_VERSION,
   586→   586→                        EaglerProfile.getUsername(),
   587→   587→                        (int)(totalTicks / 20),
   588→   588→                        (int)totalFrames);
   589→   589→        }
